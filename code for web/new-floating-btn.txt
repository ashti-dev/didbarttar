function wsbp_get_settings() {
    $defaults = wsbp_defaults();
    $saved    = get_option( WSBP_OPTION_KEY );
    if ( ! is_array( $saved ) ) return $defaults;
    $settings = wp_parse_args( $saved, $defaults );
    if ( ! isset( $settings['items'] ) || ! is_array( $settings['items'] ) || empty( $settings['items'] ) ) {
        $settings['items'] = wsbp_default_items();
    }
    return $settings;
}

add_action( 'admin_init', 'wsbp_handle_save' );
function wsbp_handle_save() {
    if ( ! isset( $_POST['wsbp_save'] ) ) return;
    if ( ! current_user_can( 'manage_options' ) ) return;
    if ( ! isset( $_POST['wsbp_nonce'] ) || ! wp_verify_nonce( wp_unslash( $_POST['wsbp_nonce'] ), 'wsbp_save_action' ) ) return;

    $items = array();
    if ( isset( $_POST['wsbp_label'] ) && is_array( $_POST['wsbp_label'] ) ) {
        foreach ( $_POST['wsbp_label'] as $i => $label ) {
            $label = sanitize_text_field( wp_unslash( $label ) );
            $desc  = isset( $_POST['wsbp_desc'][ $i ] ) ? sanitize_text_field( wp_unslash( $_POST['wsbp_desc'][ $i ] ) ) : '';
            $url   = isset( $_POST['wsbp_url'][ $i ] ) ? esc_url_raw( wp_unslash( $_POST['wsbp_url'][ $i ] ) ) : '';
            $icon  = isset( $_POST['wsbp_icon'][ $i ] ) ? sanitize_text_field( wp_unslash( $_POST['wsbp_icon'][ $i ] ) ) : 'link';
            $color = isset( $_POST['wsbp_color'][ $i ] ) ? sanitize_hex_color( wp_unslash( $_POST['wsbp_color'][ $i ] ) ) : '#0D9488';
            if ( $label === '' && $url === '' ) continue;
            $items[] = array(
                'label' => $label,
                'desc'  => $desc,
                'url'   => $url,
                'icon'  => $icon,
                'color' => $color ? $color : '#0D9488',
            );
        }
    }
    if ( empty( $items ) ) {
        $items = wsbp_default_items();
    }

    $theme       = isset( $_POST['wsbp_theme'] ) ? sanitize_text_field( wp_unslash( $_POST['wsbp_theme'] ) ) : 'light';
    $side        = isset( $_POST['wsbp_side'] ) ? sanitize_text_field( wp_unslash( $_POST['wsbp_side'] ) ) : 'right';
    $button_size = isset( $_POST['wsbp_button_size'] ) ? sanitize_text_field( wp_unslash( $_POST['wsbp_button_size'] ) ) : 'medium';
    $launcher_icon = isset( $_POST['wsbp_launcher_icon'] ) ? sanitize_text_field( wp_unslash( $_POST['wsbp_launcher_icon'] ) ) : 'support';

    $settings = array(
        'items'          => $items,
        'panel_title'    => isset( $_POST['wsbp_panel_title'] ) ? sanitize_text_field( wp_unslash( $_POST['wsbp_panel_title'] ) ) : '',
        'panel_subtitle' => isset( $_POST['wsbp_panel_subtitle'] ) ? sanitize_text_field( wp_unslash( $_POST['wsbp_panel_subtitle'] ) ) : '',
        'launcher_icon'  => $launcher_icon,
        'launcher_text'  => isset( $_POST['wsbp_launcher_text'] ) ? sanitize_text_field( wp_unslash( $_POST['wsbp_launcher_text'] ) ) : '',
        'theme'          => in_array( $theme, array( 'light', 'dark', 'auto' ), true ) ? $theme : 'light',
        'side'           => $side === 'left' ? 'left' : 'right',
        'color_start'    => sanitize_hex_color( wp_unslash( $_POST['wsbp_color_start'] ?? '' ) ) ?: '#0D9488',
        'color_end'      => sanitize_hex_color( wp_unslash( $_POST['wsbp_color_end'] ?? '' ) ) ?: '#0EA5A4',
        'show_desc'      => isset( $_POST['wsbp_show_desc'] ) ? 1 : 0,
        'show_pulse'     => isset( $_POST['wsbp_show_pulse'] ) ? 1 : 0,
        'show_tooltip'   => isset( $_POST['wsbp_show_tooltip'] ) ? 1 : 0,
        'button_size'    => in_array( $button_size, array( 'small', 'medium', 'large' ), true ) ? $button_size : 'medium',
        'desktop_right'  => isset( $_POST['wsbp_desktop_right'] ) ? intval( $_POST['wsbp_desktop_right'] ) : 28,
        'desktop_bottom' => isset( $_POST['wsbp_desktop_bottom'] ) ? intval( $_POST['wsbp_desktop_bottom'] ) : 28,
        'mobile_right'   => isset( $_POST['wsbp_mobile_right'] ) ? intval( $_POST['wsbp_mobile_right'] ) : 16,
        'mobile_bottom'  => isset( $_POST['wsbp_mobile_bottom'] ) ? intval( $_POST['wsbp_mobile_bottom'] ) : 16,
        'z_index'        => isset( $_POST['wsbp_z_index'] ) ? intval( $_POST['wsbp_z_index'] ) : 999999999,
    );

    update_option( WSBP_OPTION_KEY, $settings );
    add_action( 'admin_notices', function(){
        echo '<div class="notice notice-success is-dismissible"><p>تنظیمات با موفقیت ذخیره شد.</p></div>';
    } );
}

    <script>
    (function(){
        var tabBtns = document.querySelectorAll('.wsbp-tab-btn');
        tabBtns.forEach(function(btn){
            btn.addEventListener('click', function(){
                var target = btn.getAttribute('data-tab');
                document.querySelectorAll('.wsbp-tab-btn').forEach(function(b){ b.classList.remove('is-active'); });
                document.querySelectorAll('.wsbp-tab-panel').forEach(function(p){ p.classList.remove('is-active'); });
                btn.classList.add('is-active');
                var panel = document.querySelector('.wsbp-tab-panel[data-tab-panel="' + target + '"]');
                if (panel) panel.classList.add('is-active');
            });
        });

        if (window.jQuery && jQuery.fn.wpColorPicker) {
            jQuery('.wsbp-color-picker').wpColorPicker({
                change: function(){ setTimeout(updatePreview, 30); }
            });
        }

        var iconOptions = <?php echo wp_json_encode( $icons ); ?>;
        function buildOptions(){
            var opts = '';
            for (var key in iconOptions) {
                opts += '<option value="' + key + '">' + iconOptions[key] + '</option>';
            }
            return opts;
        }
        document.getElementById('wsbp-add-row').addEventListener('click', function(){
            var tbody = document.getElementById('wsbp-rows');
            var tr = document.createElement('tr');
            tr.innerHTML =
                '<td class="wsbp-reorder"><button type="button" class="button wsbp-move-up" title="بالا">▲</button><button type="button" class="button wsbp-move-down" title="پایین">▼</button></td>' +
                '<td><input type="text" name="wsbp_label[]" class="regular-text" placeholder="عنوان"></td>' +
                '<td><input type="text" name="wsbp_desc[]" class="regular-text" placeholder="توضیح کوتاه"></td>' +
                '<td><select name="wsbp_icon[]">' + buildOptions() + '</select></td>' +
                '<td><input type="text" name="wsbp_url[]" class="regular-text" placeholder="https://..."></td>' +
                '<td><input type="text" class="wsbp-color-picker" name="wsbp_color[]" value="#0D9488"></td>' +
                '<td><button type="button" class="button wsbp-remove-row">حذف</button></td>';
            tbody.appendChild(tr);
            if (window.jQuery && jQuery.fn.wpColorPicker) {
                jQuery(tr).find('.wsbp-color-picker').wpColorPicker();
            }
        });

        document.getElementById('wsbp-rows').addEventListener('click', function(e){
            var tr = e.target.closest('tr');
            if (!tr) return;
            if (e.target.classList.contains('wsbp-remove-row')) {
                tr.remove();
            } else if (e.target.classList.contains('wsbp-move-up')) {
                var prev = tr.previousElementSibling;
                if (prev) tr.parentNode.insertBefore(tr, prev);
            } else if (e.target.classList.contains('wsbp-move-down')) {
                var next = tr.nextElementSibling;
                if (next) tr.parentNode.insertBefore(next, tr);
            }
        });

        function updatePreview(){
            var c1 = document.querySelector('input[name="wsbp_color_start"]');
            var c2 = document.querySelector('input[name="wsbp_color_end"]');
            var title = document.querySelector('input[name="wsbp_panel_title"]');
            var subtitle = document.querySelector('input[name="wsbp_panel_subtitle"]');
            var box = document.querySelector('.wsbp-preview-box');
            if (c1 && c2 && box) {
                box.style.setProperty('--pv1', c1.value || '#0D9488');
                box.style.setProperty('--pv2', c2.value || '#0EA5A4');
            }
            var t = document.getElementById('wsbp-pv-title');
            var s2 = document.getElementById('wsbp-pv-subtitle');
            if (t && title) t.textContent = title.value || 'در ارتباط باشیم';
            if (s2 && subtitle) s2.textContent = subtitle.value || 'یکی از راه‌ها رو انتخاب کن';
        }
        document.querySelectorAll('input[name="wsbp_panel_title"], input[name="wsbp_panel_subtitle"], input[name="wsbp_color_start"], input[name="wsbp_color_end"]').forEach(function(el){
            el.addEventListener('input', updatePreview);
        });
        updatePreview();
    })();
    </script>
