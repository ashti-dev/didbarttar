    public function register_settings() {

    public function sanitize_settings( $input ) {
        if ( ! current_user_can( 'manage_options' ) ) {
            return $this->get_settings();
        }
        $d = $this->defaults();
        $out = array();

        $out['enabled']        = ! empty( $input['enabled'] ) ? 1 : 0;
        $out['show_text']      = ! empty( $input['show_text'] ) ? 1 : 0;
        $out['hide_on_mobile'] = ! empty( $input['hide_on_mobile'] ) ? 1 : 0;
        $out['use_webfont']    = ! empty( $input['use_webfont'] ) ? 1 : 0;
        $out['backdrop_blur']  = ! empty( $input['backdrop_blur'] ) ? 1 : 0;
        $out['card_shadow']    = ! empty( $input['card_shadow'] ) ? 1 : 0;
        $out['colorful_style'] = ! empty( $input['colorful_style'] ) ? 1 : 0;

        foreach ( array( 'bg_color', 'primary_color', 'secondary_color', 'text_color', 'card_bg' ) as $c ) {
            $val = isset( $input[ $c ] ) ? sanitize_hex_color( $input[ $c ] ) : '';
            $out[ $c ] = $val ? $val : $d[ $c ];
        }

        $allowed_styles = array( 'ring', 'dual_ring', 'pulse' );
        $out['spinner_style'] = ( isset( $input['spinner_style'] ) && in_array( $input['spinner_style'], $allowed_styles, true ) )
            ? $input['spinner_style'] : $d['spinner_style'];

        $out['logo_url']     = isset( $input['logo_url'] ) ? esc_url_raw( $input['logo_url'] ) : '';
        $out['loading_text'] = isset( $input['loading_text'] ) ? sanitize_text_field( $input['loading_text'] ) : $d['loading_text'];

        foreach ( array( 'logo_size', 'font_size', 'min_time', 'max_wait', 'fade_duration', 'z_index', 'card_radius', 'card_padding', 'backdrop_opacity' ) as $n ) {
            $out[ $n ] = isset( $input[ $n ] ) ? absint( $input[ $n ] ) : $d[ $n ];
        }
        // محدودسازی مقادیر عددی در بازه‌ی منطقی
        $out['logo_size']        = min( max( $out['logo_size'], 24 ), 300 );
        $out['font_size']        = min( max( $out['font_size'], 10 ), 32 );
        $out['min_time']         = min( max( $out['min_time'], 0 ), 3000 );
        $out['max_wait']         = min( max( $out['max_wait'], 2000 ), 20000 );
        $out['fade_duration']    = min( max( $out['fade_duration'], 100 ), 2000 );
        $out['z_index']          = min( max( $out['z_index'], 1000 ), 2147483000 );
        $out['card_radius']      = min( max( $out['card_radius'], 0 ), 60 );
        $out['card_padding']     = min( max( $out['card_padding'], 16 ), 100 );
        $out['backdrop_opacity'] = min( max( $out['backdrop_opacity'], 10 ), 100 );

        return $out;
    }

    /* ---------------------------------------------------------
     * تزریق لودینگ در فرانت سایت
     * --------------------------------------------------------- */
    public function render_loader() {
        if ( is_admin() || wp_doing_ajax() || is_feed() || ( defined( 'REST_REQUEST' ) && REST_REQUEST ) ) {
            return;
        }
        $s = $this->get_settings();
        if ( empty( $s['enabled'] ) ) {
            return;
        }
        echo $this->build_loader_html(); // phpcs:ignore -- خروجی از قبل با esc_* پاک‌سازی شده
    }

    private function hex_to_rgb( $hex ) {
        $hex = ltrim( (string) $hex, '#' );
        if ( strlen( $hex ) === 3 ) {
            $hex = $hex[0] . $hex[0] . $hex[1] . $hex[1] . $hex[2] . $hex[2];
        }
        if ( strlen( $hex ) !== 6 ) {
            return '255,255,255';
        }
        $r = hexdec( substr( $hex, 0, 2 ) );
        $g = hexdec( substr( $hex, 2, 2 ) );
        $b = hexdec( substr( $hex, 4, 2 ) );
        return "$r,$g,$b";
    }

    private function build_loader_html() {
        $s = $this->get_settings();

        $bg_rgb     = $this->hex_to_rgb( $s['bg_color'] );
        $bg_opacity = round( absint( $s['backdrop_opacity'] ) / 100, 2 );
        $card_bg    = esc_attr( $s['card_bg'] );
        $card_radius= absint( $s['card_radius'] );
        $card_padding = absint( $s['card_padding'] );
        $card_shadow  = ! empty( $s['card_shadow'] );
        $blur         = ! empty( $s['backdrop_blur'] );
        $colorful     = ! empty( $s['colorful_style'] );

        $primary = esc_attr( $s['primary_color'] );
        $secondary = esc_attr( $s['secondary_color'] );
        $primary_rgb   = $this->hex_to_rgb( $s['primary_color'] );
        $secondary_rgb = $this->hex_to_rgb( $s['secondary_color'] );
        $text_color = esc_attr( $s['text_color'] );
        $font_size  = absint( $s['font_size'] );
        $logo_size  = absint( $s['logo_size'] );
        $z_index    = absint( $s['z_index'] );
        $min_time   = absint( $s['min_time'] );
        $max_wait   = absint( $s['max_wait'] );
        $fade       = absint( $s['fade_duration'] );
        $style      = esc_attr( $s['spinner_style'] );
        $logo_url   = esc_url( $s['logo_url'] );
        $has_logo   = ! empty( $s['logo_url'] );
        $show_text  = ! empty( $s['show_text'] );
        $loading_text = esc_html( $s['loading_text'] );
        $hide_mobile  = ! empty( $s['hide_on_mobile'] );

        // پیش‌فرض: فونت از خود سایت به ارث برده می‌شود (inherit)، نه یک فونت تحمیلی.
        $font_import = '';
        $font_family = 'inherit';
        if ( ! empty( $s['use_webfont'] ) ) {
            $font_import = "@import url('https://fonts.googleapis.com/css2?family=Vazirmatn:wght@400;600&display=swap');";
            $font_family = "'Vazirmatn', inherit";
        }

        $spinner_html = '';
        if ( 'pulse' === $style ) {
            if ( $has_logo ) {
                $spinner_html = '<img src="' . $logo_url . '" alt="" class="wds-pulse-logo" style="width:clamp(36px, 20vw, ' . $logo_size . 'px);height:clamp(36px, 20vw, ' . $logo_size . 'px);object-fit:contain;" />';
            } else {
                $spinner_html = '<div class="wds-pulse-dot" style="width:clamp(36px, 20vw, ' . $logo_size . 'px);height:clamp(36px, 20vw, ' . $logo_size . 'px);background:' . $primary . ';"></div>';

(function(){

    function __wdsDots(){
        var el = document.querySelector('.wds-dots-js');
        if(!el) return;

    function __wdsHide(){
        if(__wdsHidden) return;
        __wdsHidden = true;
        var el = document.getElementById('wds-loader-overlay');
        if(!el) return;
        el.classList.add('wds-hide');
        setTimeout(function(){
            if(el && el.parentNode){ el.parentNode.removeChild(el); }
        }, __wdsFade + 60);
    }

    function __wdsReveal(){
        var elapsed = Date.now() - __wdsStart;
        var wait = Math.max(__wdsMin - elapsed, 0);
        setTimeout(__wdsHide, wait);
    }

    if (document.readyState === 'complete') {
        __wdsReveal();
    } else {
        window.addEventListener('load', __wdsReveal);

})();
