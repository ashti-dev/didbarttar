if (!function_exists('awc_wc_default_settings')) {
    function awc_wc_default_settings() {
        return array(
            'enabled'        => 1,
            'currency_label' => 'تومان',
            'tiers'          => array(
                array('amount' => 3000000, 'percent' => 5),
                array('amount' => 5000000, 'percent' => 10),
                array('amount' => 7000000, 'percent' => 15),
            ),
        );
    }
}

if (!function_exists('awc_wc_get_settings')) {
    function awc_wc_get_settings() {
        $saved    = get_option('awc_tiered_discount_settings');
        $defaults = awc_wc_default_settings();

        if (!is_array($saved)) {
            return $defaults;
        }

        $saved['enabled']        = isset($saved['enabled']) ? (bool) $saved['enabled'] : true;
        $saved['currency_label'] = (!empty($saved['currency_label'])) ? $saved['currency_label'] : $defaults['currency_label'];
        $saved['tiers']          = (!empty($saved['tiers']) && is_array($saved['tiers'])) ? $saved['tiers'] : $defaults['tiers'];

        // اطمینان از مرتب بودن صعودی پله‌ها بر اساس مبلغ
        usort($saved['tiers'], function ($a, $b) {
            return $a['amount'] <=> $b['amount'];
        });

        return $saved;
    }
}

}

function awc_wc_sanitize_settings($input) {
    $defaults = awc_wc_default_settings();
    $output   = array();

    $output['enabled']        = isset($input['enabled']) ? 1 : 0;
    $output['currency_label'] = isset($input['currency_label']) ? sanitize_text_field($input['currency_label']) : $defaults['currency_label'];

    $tiers = array();
    if (isset($input['tier_amount']) && is_array($input['tier_amount'])) {
        foreach ($input['tier_amount'] as $i => $amount) {
            $amount  = (float) preg_replace('/[^0-9.]/', '', (string) $amount);
            $percent = isset($input['tier_percent'][$i]) ? (float) preg_replace('/[^0-9.]/', '', (string) $input['tier_percent'][$i]) : 0;

            if ($amount > 0 && $percent > 0 && $percent <= 100) {
                $tiers[] = array(
                    'amount'  => $amount,
                    'percent' => $percent,
                );
            }
        }
    }

    usort($tiers, function ($a, $b) {
        return $a['amount'] <=> $b['amount'];
    });

    $output['tiers'] = !empty($tiers) ? array_values($tiers) : $defaults['tiers'];

    add_settings_error('awc_tiered_discount_settings', 'awc_settings_saved', 'تنظیمات با موفقیت ذخیره شد.', 'success');

    return $output;
}

add_action('woocommerce_cart_calculate_fees', 'awc_wc_apply_tiered_discount', 20, 1);
function awc_wc_apply_tiered_discount($cart) {
    if (is_admin() && !defined('DOING_AJAX')) return;
    if (!$cart || $cart->is_empty()) return;

    $settings = awc_wc_get_settings();
    if (empty($settings['enabled']) || empty($settings['tiers'])) return;

    $subtotal = (float) $cart->get_subtotal();
    $percent  = awc_wc_get_applicable_percent($subtotal, $settings['tiers']);

    if ($percent > 0) {
        $discount_amount = $subtotal * ($percent / 100);
        if ($discount_amount > 0) {
            $cart->add_fee(
                sprintf('تخفیف پله‌ای (%s%%)', awc_wc_format_number($percent)),
                -$discount_amount,
                false
            );
        }
    }
}

function awc_wc_get_applicable_percent($subtotal, $tiers) {
    $percent = 0;
    foreach ($tiers as $tier) {
        if ($subtotal >= $tier['amount'] && $tier['percent'] > $percent) {
            $percent = $tier['percent'];
        }
    }
    return $percent;
}

function awc_wc_format_number($number) {
    // حذف صفرهای اضافه اعشار (5.0 -> 5)
    return (floor($number) == $number) ? number_format($number, 0) : rtrim(rtrim(number_format($number, 2), '0'), '.');
}

add_action('woocommerce_before_cart', 'awc_wc_render_discount_notice');
add_action('woocommerce_before_checkout_form', 'awc_wc_render_discount_notice');
function awc_wc_render_discount_notice() {
    if (!function_exists('WC') || !WC()->cart) return;

    $settings = awc_wc_get_settings();
    if (empty($settings['enabled']) || empty($settings['tiers'])) return;

    $tiers          = array_values($settings['tiers']);
    $currency_label = esc_html($settings['currency_label']);
    $subtotal       = (float) WC()->cart->get_subtotal();

    $current = null;
    $next    = null;

    foreach ($tiers as $tier) {
        if ($subtotal >= $tier['amount']) {
            $current = $tier;
        } elseif ($next === null) {
            $next = $tier;
        }
    }

    if (!$current && !$next) return;

    $state   = 'progress';
    $percent = 0;
    $message = '';

    if (!$current && $next) {
        $remaining = max(0, $next['amount'] - $subtotal);
        $percent   = $next['amount'] > 0 ? min(100, ($subtotal / $next['amount']) * 100) : 0;
        $state     = 'progress';
        $message   = sprintf(
            '🛒 فقط <strong>%s %s</strong> دیگر مانده تا <strong>%s%%</strong> تخفیف فعال شود!',
            awc_wc_format_number($remaining),
            $currency_label,
            awc_wc_format_number($next['percent'])
        );
    } elseif ($current && $next) {
        $remaining = max(0, $next['amount'] - $subtotal);
        $range     = $next['amount'] - $current['amount'];
        $percent   = $range > 0 ? min(100, (($subtotal - $current['amount']) / $range) * 100) : 100;
        $state     = 'success';
        $message   = sprintf(
            '✅ تبریک! <strong>%s%%</strong> تخفیف گرفتید. فقط <strong>%s %s</strong> دیگر مانده تا <strong>%s%%</strong> تخفیف!',
            awc_wc_format_number($current['percent']),
            awc_wc_format_number($remaining),
            $currency_label,
            awc_wc_format_number($next['percent'])
        );
    } else {
        $percent = 100;
        $state   = 'max';
        $message = sprintf(
            '🎉 هورااا! شما به بالاترین سطح تخفیف (<strong>%s%%</strong>) رسیدید!',
            awc_wc_format_number($current['percent'])
        );
    }

    echo '<div class="awc-notice awc-notice-' . esc_attr($state) . '">
            <div class="awc-notice-icon">' . ($state === 'max' ? '🎉' : ($state === 'success' ? '✅' : '🛒')) . '</div>
            <div class="awc-notice-body">
                <div class="awc-notice-text">' . wp_kses_post($message) . '</div>
                <div class="awc-progress-track">
                    <div class="awc-progress-fill" style="width:' . esc_attr($percent) . '%;"></div>
                </div>
            </div>
          </div>';
}
