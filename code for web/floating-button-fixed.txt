function wafm_get_settings() {
    $defaults = array(
        'items'      => wafm_default_items(),
        'main_start' => '#7F53FF',
        'main_end'   => '#647DEE',
        'position'   => array( 'right' => 24, 'bottom' => 24 ),
    );
    $saved = get_option( 'wafm_settings' );
    if ( ! is_array( $saved ) ) return $defaults;
    return wp_parse_args( $saved, $defaults );
}

               ولی اگر عدد وارد شده از عرض/ارتفاع صفحه در گوشی بزرگتر بود، به‌صورت خودکار به داخل صفحه محدود بشه

add_action( 'admin_init', 'wafm_handle_save' );
function wafm_handle_save() {
    if ( ! isset( $_POST['wafm_save'] ) ) return;
    if ( ! current_user_can( 'manage_options' ) ) return;
    if ( ! isset( $_POST['wafm_nonce'] ) || ! wp_verify_nonce( $_POST['wafm_nonce'], 'wafm_save_action' ) ) return;

    $items = array();
    if ( isset( $_POST['wafm_label'] ) && is_array( $_POST['wafm_label'] ) ) {
        foreach ( $_POST['wafm_label'] as $i => $label ) {
            $label = sanitize_text_field( $label );
            $url   = isset( $_POST['wafm_url'][ $i ] ) ? esc_url_raw( $_POST['wafm_url'][ $i ] ) : '';
            $icon  = isset( $_POST['wafm_icon'][ $i ] ) ? sanitize_text_field( $_POST['wafm_icon'][ $i ] ) : 'link';
            $color = isset( $_POST['wafm_color'][ $i ] ) ? sanitize_hex_color( $_POST['wafm_color'][ $i ] ) : '#7F53FF';
            if ( $label === '' && $url === '' ) continue;
            $items[] = array(
                'label' => $label,
                'url'   => $url,
                'icon'  => $icon,
                'color' => $color ? $color : '#7F53FF',
            );
        }
    }

    $settings = array(
        'items'      => $items,
        'main_start' => sanitize_hex_color( $_POST['wafm_main_start'] ) ?: '#7F53FF',
        'main_end'   => sanitize_hex_color( $_POST['wafm_main_end'] ) ?: '#647DEE',
        'position'   => array(
            'right'  => intval( $_POST['wafm_right'] ),
            'bottom' => intval( $_POST['wafm_bottom'] ),
        ),
    );

    update_option( 'wafm_settings', $settings );
    add_action( 'admin_notices', function(){
        echo '<div class="notice notice-success is-dismissible"><p>تنظیمات دکمه شناور با موفقیت ذخیره شد.</p></div>';
    } );
}
