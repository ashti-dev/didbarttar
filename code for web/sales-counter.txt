if ( ! defined( 'IWSC_LOADED' ) ) {

	define( 'IWSC_LOADED', true );

	add_action( 'plugins_loaded', 'iwsc_bootstrap' );

	function iwsc_bootstrap() {
		if ( ! class_exists( 'WooCommerce' ) ) {
			return;
		}

		add_action( 'woocommerce_after_add_to_cart_button', 'iwsc_render_sales_counter_desktop', 15 );


		add_action( 'wp_footer', 'iwsc_render_sales_counter_mobile' );

		add_action( 'admin_menu', 'iwsc_add_settings_menu' );
	}

	function iwsc_get_defaults() {
		return array(
			'enable'        => '1',
			'start_count'   => 100,
			'text_template' => '{count} فروش موفق برای این محصول',
			'bg_color'      => '#eaf7ec',
			'text_color'    => '#1e7e34',
			'border_color'  => '#bfe6c4',
			'font_size'     => 14,
			'border_radius' => 10,
			'icon'          => '🛒',
		);
	}


	function iwsc_get_option( $key ) {
		$defaults = iwsc_get_defaults();
		$stored   = get_option( 'iwsc_' . $key, null );
		return ( $stored === null || $stored === '' ) ? $defaults[ $key ] : $stored;
	}


	function iwsc_render_sales_counter_desktop() {
		iwsc_render_sales_counter( 'desktop' );
	}

	function iwsc_render_sales_counter_mobile() {

		if ( ! function_exists( 'is_product' ) || ! is_product() ) {
			return;
		}

		if ( iwsc_get_option( 'enable' ) !== '1' ) {
			return;
		}

		global $product;

		if ( ! is_a( $product, 'WC_Product' ) ) {
			return;
		}

		$start_count = absint( iwsc_get_option( 'start_count' ) );
		$real_sales  = absint( $product->get_total_sales() );
		$total       = $start_count + $real_sales;

		$template = iwsc_get_option( 'text_template' );
		$text     = str_replace( '{count}', number_format_i18n( $total ), $template );
		$icon     = iwsc_get_option( 'icon' );

		iwsc_print_inline_style();

		echo '<div id="iwsc-mobile-badge" class="iwsc-wrap iwsc-wrap-mobile" style="position:absolute;left:-9999px;top:-9999px;visibility:hidden;">';
		echo '<div class="iwsc-badge">';
		if ( $icon !== 'none' && $icon !== '' ) {
			echo '<span class="iwsc-icon" aria-hidden="true">' . esc_html( $icon ) . '</span>';
		}
		echo '<span class="iwsc-text">' . esc_html( $text ) . '</span>';
		echo '</div>';
		echo '</div>';
		?>
		<script>
		(function () {
			var badge = document.getElementById( 'iwsc-mobile-badge' );
			if ( ! badge ) { return; }


			var titleEl = document.querySelector(
				'.summary .product_title, .product_title, .summary h1, h1.entry-title'
			);
			var summaryEl = document.querySelector( '.summary.entry-summary, div.summary' );

			if ( titleEl && titleEl.parentNode ) {
				titleEl.insertAdjacentElement( 'afterend', badge );
			} else if ( summaryEl ) {
				summaryEl.insertBefore( badge, summaryEl.firstChild );
			}

	
			badge.style.position   = '';
			badge.style.left       = '';
			badge.style.top        = '';
			badge.style.visibility = '';
		})();
		</script>
		<?php
	}

	function iwsc_render_sales_counter( $position = 'desktop' ) {

		if ( iwsc_get_option( 'enable' ) !== '1' ) {
			return;
		}

		global $product;

		if ( ! is_a( $product, 'WC_Product' ) ) {
			return;
		}

		$start_count = absint( iwsc_get_option( 'start_count' ) );
		$real_sales  = absint( $product->get_total_sales() );
		$total       = $start_count + $real_sales;

		$template = iwsc_get_option( 'text_template' );
		$text     = str_replace( '{count}', number_format_i18n( $total ), $template );
		$icon     = iwsc_get_option( 'icon' );

		iwsc_print_inline_style();

		$position_class = ( $position === 'mobile' ) ? 'iwsc-wrap-mobile' : 'iwsc-wrap-desktop';

		echo '<div class="iwsc-wrap ' . esc_attr( $position_class ) . '">';
		echo '<div class="iwsc-badge">';
		if ( $icon !== 'none' && $icon !== '' ) {
			echo '<span class="iwsc-icon" aria-hidden="true">' . esc_html( $icon ) . '</span>';
		}
		echo '<span class="iwsc-text">' . esc_html( $text ) . '</span>';
		echo '</div>';
		echo '</div>';
	}


	function iwsc_print_inline_style() {
		static $printed = false;
		if ( $printed ) {
			return;
		}
		$printed = true;

		$bg     = sanitize_hex_color( iwsc_get_option( 'bg_color' ) );
		$color  = sanitize_hex_color( iwsc_get_option( 'text_color' ) );
		$border = sanitize_hex_color( iwsc_get_option( 'border_color' ) );
		$fsize  = absint( iwsc_get_option( 'font_size' ) );
		$radius = absint( iwsc_get_option( 'border_radius' ) );

		$bg     = $bg ? $bg : '#eaf7ec';
		$color  = $color ? $color : '#1e7e34';
		$border = $border ? $border : '#bfe6c4';
		$fsize  = $fsize ? $fsize : 14;
		?>
		<style id="iwsc-style">
			.iwsc-wrap{
				width:100%;
				max-width:100%;
				flex:1 1 100%;
				clear:both;
				box-sizing:border-box;
				margin:14px 0 0 0;
			}

	
			.iwsc-wrap-desktop{ display:block; }
			.iwsc-wrap-mobile{ display:none; }

			
			@media (max-width:768px){
				.iwsc-wrap-desktop{ display:none; }
				.iwsc-wrap-mobile{ display:block; margin:10px 0 16px 0; }
			}

			.iwsc-badge{
				display:flex;
				align-items:center;
				justify-content:center;
				gap:8px;
				width:100%;
				box-sizing:border-box;
				padding:10px 14px;
				background-color:<?php echo esc_html( $bg ); ?>;
				color:<?php echo esc_html( $color ); ?>;
				border:1px solid <?php echo esc_html( $border ); ?>;
				border-radius:<?php echo (int) $radius; ?>px;
				font-family:inherit;
				font-size:<?php echo (int) $fsize; ?>px;
				font-weight:600;
				line-height:1.7;
				text-align:center;
				direction:rtl;
				unicode-bidi:plain-text;
				word-break:break-word;
			}
			.iwsc-icon{
				font-size:1.15em;
				line-height:1;
				flex-shrink:0;
			}
			.iwsc-text{
				flex-shrink:1;
			}
			@media (max-width:480px){
				.iwsc-badge{
					font-size:<?php echo max( 12, (int) $fsize - 1 ); ?>px;
					padding:8px 10px;
					gap:6px;
				}
			}
		</style>
		<?php
	}

	function iwsc_add_settings_menu() {
		if ( ! current_user_can( 'manage_woocommerce' ) ) {
			return;
		}
		add_submenu_page(
			'woocommerce',
			'شمارنده فروش',
			'شمارنده فروش',
			'manage_woocommerce',
			'iwsc-settings',
			'iwsc_render_settings_page'
		);
	}

	function iwsc_render_settings_page() {

		if ( ! current_user_can( 'manage_woocommerce' ) ) {
			return;
		}

		if (
			isset( $_POST['iwsc_save'] ) &&
			isset( $_POST['iwsc_nonce'] ) &&
			wp_verify_nonce( sanitize_text_field( wp_unslash( $_POST['iwsc_nonce'] ) ), 'iwsc_save_settings' )
		) {
			update_option( 'iwsc_enable', isset( $_POST['iwsc_enable'] ) ? '1' : '0' );
			update_option( 'iwsc_start_count', absint( $_POST['iwsc_start_count'] ?? 0 ) );
			update_option( 'iwsc_text_template', sanitize_text_field( wp_unslash( $_POST['iwsc_text_template'] ?? '{count} فروش موفق برای این محصول' ) ) );
			update_option( 'iwsc_bg_color', sanitize_hex_color( wp_unslash( $_POST['iwsc_bg_color'] ?? '' ) ) );
			update_option( 'iwsc_text_color', sanitize_hex_color( wp_unslash( $_POST['iwsc_text_color'] ?? '' ) ) );
			update_option( 'iwsc_border_color', sanitize_hex_color( wp_unslash( $_POST['iwsc_border_color'] ?? '' ) ) );
			update_option( 'iwsc_font_size', absint( $_POST['iwsc_font_size'] ?? 14 ) );
			update_option( 'iwsc_border_radius', absint( $_POST['iwsc_border_radius'] ?? 10 ) );
			update_option( 'iwsc_icon', sanitize_text_field( wp_unslash( $_POST['iwsc_icon'] ?? '🛒' ) ) );

			echo '<div class="notice notice-success is-dismissible"><p>تنظیمات با موفقیت ذخیره شد.</p></div>';
		}

		$enable        = iwsc_get_option( 'enable' );
		$start_count   = iwsc_get_option( 'start_count' );
		$text_template = iwsc_get_option( 'text_template' );
		$bg_color      = iwsc_get_option( 'bg_color' );
		$text_color    = iwsc_get_option( 'text_color' );
		$border_color  = iwsc_get_option( 'border_color' );
		$font_size     = iwsc_get_option( 'font_size' );
		$border_radius = iwsc_get_option( 'border_radius' );
		$icon          = iwsc_get_option( 'icon' );

		$icon_options = array(
			'none' => 'بدون آیکون',
			'🛒'   => '🛒 سبد خرید',
			'✅'   => '✅ تیک تایید',
			'🔥'   => '🔥 آتش (پرفروش)',
			'⭐'   => '⭐ ستاره',
			'👍'   => '👍 لایک',
		);
		?>
		<div class="wrap" dir="rtl" style="max-width:900px;">
			<h1 style="margin-bottom:4px;">تنظیمات شمارنده فروش موفق</h1>
			<p style="color:#666;">
				این ابزار تعداد فروش موفق را نمایش می‌دهد؛ در دسکتاپ زیر دکمه «افزودن به سبد خرید»
				و در موبایل زیر عنوان محصول (چون در موبایل نوار چسبان پایین صفحه مانع نمایش زیر دکمه می‌شود).
			</p>

			<form method="post">
				<?php wp_nonce_field( 'iwsc_save_settings', 'iwsc_nonce' ); ?>

				<table class="form-table" role="presentation">
					<tr>
						<th scope="row">فعال‌سازی نمایش</th>
						<td>
							<label>
								<input type="checkbox" name="iwsc_enable" value="1" <?php checked( $enable, '1' ); ?> />
								نمایش شمارنده در صفحه محصولات
							</label>
						</td>
					</tr>

					<tr>
						<th scope="row"><label for="iwsc_start_count">شروع شمارنده</label></th>
						<td>
							<input type="number" min="0" step="1" id="iwsc_start_count" name="iwsc_start_count"
								value="<?php echo esc_attr( $start_count ); ?>" class="regular-text" />
							<p class="description">
								این عدد به تعداد فروش واقعی هر محصول (از ووکامرس) اضافه می‌شود.
								مثلاً اگر اینجا ۱۰۰ باشد و محصول ۳ فروش واقعی داشته باشد، عدد نمایشی ۱۰۳ خواهد بود.
							</p>
						</td>
					</tr>

					<tr>
						<th scope="row"><label for="iwsc_text_template">قالب متن</label></th>
						<td>
							<input type="text" id="iwsc_text_template" name="iwsc_text_template"
								value="<?php echo esc_attr( $text_template ); ?>" class="regular-text" dir="rtl" />
							<p class="description">از <code>{count}</code> برای جای‌گذاری عدد استفاده کنید. مثال: <code>{count} فروش موفق برای این محصول</code></p>
						</td>
					</tr>

					<tr>
						<th scope="row"><label for="iwsc_icon">آیکون</label></th>
						<td>
							<select id="iwsc_icon" name="iwsc_icon">
								<?php foreach ( $icon_options as $val => $label ) : ?>
									<option value="<?php echo esc_attr( $val ); ?>" <?php selected( $icon, $val ); ?>>
										<?php echo esc_html( $label ); ?>
									</option>
								<?php endforeach; ?>
							</select>
						</td>
					</tr>

					<tr>
						<th scope="row"><label for="iwsc_bg_color">رنگ پس‌زمینه</label></th>
						<td><input type="color" id="iwsc_bg_color" name="iwsc_bg_color" value="<?php echo esc_attr( $bg_color ); ?>" /></td>
					</tr>

					<tr>
						<th scope="row"><label for="iwsc_text_color">رنگ متن</label></th>
						<td><input type="color" id="iwsc_text_color" name="iwsc_text_color" value="<?php echo esc_attr( $text_color ); ?>" /></td>
					</tr>

					<tr>
						<th scope="row"><label for="iwsc_border_color">رنگ حاشیه</label></th>
						<td><input type="color" id="iwsc_border_color" name="iwsc_border_color" value="<?php echo esc_attr( $border_color ); ?>" /></td>
					</tr>

				<div id="iwsc-preview-wrap" style="max-width:400px;">
					<div id="iwsc-preview-badge" style="display:flex;align-items:center;justify-content:center;gap:8px;

					<button type="submit" name="iwsc_save" class="button button-primary">ذخیره تنظیمات</button>
				</p>
			</form>
			
							<p style="color:#fff;background:red;font-size:16px;text-align:center;Padding:20px;">ساخته شده توسط علیرضا رحمتی | اینستاگرام : alireza.webcode | تلگرام : alireza_webcode</p>

				var textEl = document.getElementById('iwsc-preview-text');
				if (!badge) return;

				var color = document.getElementById('iwsc_text_color').value;
				var border = document.getElementById('iwsc_border_color').value;
				var fsize = document.getElementById('iwsc_font_size').value || 14;
				var radius = document.getElementById('iwsc_border_radius').value || 10;
				var icon = document.getElementById('iwsc_icon').value;
				var template = document.getElementById('iwsc_text_template').value || '{count} فروش موفق برای این محصول';
				var start = parseInt(document.getElementById('iwsc_start_count').value || 0, 10);

				badge.style.backgroundColor = bg;
				badge.style.color = color;

				badge.style.fontSize = fsize + 'px';

				iconEl.textContent = (icon === 'none') ? '' : icon;

			var ids = ['iwsc_bg_color','iwsc_text_color','iwsc_border_color','iwsc_font_size','iwsc_border_radius','iwsc_icon','iwsc_text_template','iwsc_start_count'];
			ids.forEach(function(id){
