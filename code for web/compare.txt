add_action('wp_ajax_aroma_get_products','aroma_get_products');
add_action('wp_ajax_nopriv_aroma_get_products','aroma_get_products');


function aroma_get_products(){

$id=intval($_POST['id']);

$product=wc_get_product($id);

if(!$product){
wp_send_json_error();
}


$cats=wp_get_post_terms(
$id,
'product_cat',
array('fields'=>'ids')
);

$args=array(

'post_type'=>'product',
'posts_per_page'=>100,

'post__not_in'=>array(
    intval($id)
),

'tax_query'=>array(

array(

'taxonomy'=>'product_cat',
'field'=>'term_id',
'terms'=>$cats

)

),

'post_status'=>'publish'

);
$q=new WP_Query($args);


$html='';


foreach($q->posts as $p){


    // حذف محصول فعلی از لیست
    if($p->ID == $id){
        continue;
    }


    $pro=wc_get_product($p->ID);


$html.='

<div class="aroma-select-product" data-id="'.$pro->get_id().'">

<img src="'.get_the_post_thumbnail_url($pro->get_id(),'thumbnail').'">

<div>
<strong>'.$pro->get_name().'</strong>
<br>
<span>'.$pro->get_price_html().'</span>
</div>

</div>';

}



wp_send_json_success($html);

}

add_action('wp_ajax_aroma_compare','aroma_compare');
add_action('wp_ajax_nopriv_aroma_compare','aroma_compare');


function aroma_compare(){

$a=wc_get_product($_POST['a']);
$b=wc_get_product($_POST['b']);


if(!$a || !$b)
wp_send_json_error();



$data=array();


$data['head']=array(

array(
$a->get_name(),
get_the_post_thumbnail_url($a->get_id(),'medium')
),

array(
$b->get_name(),
get_the_post_thumbnail_url($b->get_id(),'medium')
)

);



$rows=array();



$rows[]=array(
'قیمت اصلی',
wc_price($a->get_regular_price()),
wc_price($b->get_regular_price())
);


$rows[]=array(
'قیمت با تخفیف',
wc_price($a->get_price()),
wc_price($b->get_price())
);


$rows[]=array(
'موجودی',
$a->is_in_stock()?'موجود':'ناموجود',
$b->is_in_stock()?'موجود':'ناموجود'
);


$attrs=array_unique(array_merge(
array_keys($a->get_attributes()),
array_keys($b->get_attributes())
));

foreach($attrs as $att){


    if(strpos($att,'pa_') === 0){

        $label = urldecode(
            wc_attribute_label($att)
        );

    }else{

        $label = urldecode($att);

    }


    $value_a = $a->get_attribute($att);
    $value_b = $b->get_attribute($att);



    $rows[]=array(

        $label ?: 'ویژگی',

        $value_a ?: '-',

        $value_b ?: '-'

    );


}


$data['rows']=$rows;


wp_send_json_success($data);


}

function beautiful_product_specs_shortcode(){

        return '';

    global $product;

    if(!$product){
        return '';
    }

    $attributes = $product->get_attributes();

    if(empty($attributes)){
        return '<p>مشخصاتی برای این محصول ثبت نشده است.</p>';
    }


    ob_start();

?>

<div class="beautiful-specs-box">

    <div class="specs-list">

    <?php foreach($attributes as $attribute):

        if($attribute->is_taxonomy()){

            $values = wc_get_product_terms(
                $product->get_id(),
                $attribute->get_name(),
                array('fields'=>'names')
            );

            $value = implode(
                '، ',
                $attribute->get_options()

            $attribute->get_name()
