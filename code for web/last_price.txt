function custom_last_price_update_shortcode() {
    if ( ! is_product() ) {
        return '';
    }

    global $product;
    if ( ! $product ) {
        $product = wc_get_product( get_the_ID() );
    }

    if ( ! $product ) {
        return '';
    }
// برنامه نویس : علیرضا رحمتی | instagram : alireza.webcode | telegram : alireza_webcode 

    $modified_date_timestamp = get_post_modified_time( 'U', false, $product->get_id() );

    if ( $modified_date_timestamp ) {

        $date_format = 'j F Y'; 

        if ( function_exists( 'jdate' ) ) {
            $last_update_date = jdate( $date_format, $modified_date_timestamp );
        } elseif ( function_exists( 'pdate' ) ) {
            $last_update_date = pdate( $date_format, $modified_date_timestamp );
        } else {
            $last_update_date = date_i18n( $date_format, $modified_date_timestamp );
        }
// برنامه نویس : علیرضا رحمتی | instagram : alireza.webcode | telegram : alireza_webcode 

        $output = '<span class="last-price-update">';
        $output .= 'آخرین بروزرسانی قیمت: ';
        $output .= '<time datetime="' . date( 'Y-m-d H:i:s', $modified_date_timestamp ) . '">';
        $output .= $last_update_date;
        $output .= '</time>';
        $output .= '</span>';

        return $output;
    }

    return '';
}
// برنامه نویس : علیرضا رحمتی | instagram : alireza.webcode | telegram : alireza_webcode 

add_shortcode( 'last_price', 'custom_last_price_update_shortcode' );
