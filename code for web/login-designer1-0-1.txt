function bld_get_options() {
    $saved = get_option( 'bld_login_options', array() );
    return wp_parse_args( $saved, bld_get_default_options() );
}

add_action( 'admin_init', 'bld_register_settings' );
function bld_register_settings() {
    register_setting( 'bld_login_settings_group', 'bld_login_options', 'bld_sanitize_options' );
}

function bld_sanitize_options( $input ) {
    $defaults = bld_get_default_options();
    $out = array();

    $out['bg_color_1']     = sanitize_hex_color( $input['bg_color_1'] ?? $defaults['bg_color_1'] ) ?: $defaults['bg_color_1'];
    $out['bg_color_2']     = sanitize_hex_color( $input['bg_color_2'] ?? $defaults['bg_color_2'] ) ?: $defaults['bg_color_2'];
    $out['animate_bg']     = ! empty( $input['animate_bg'] ) ? 1 : 0;

    $out['form_bg_color']  = sanitize_hex_color( $input['form_bg_color'] ?? $defaults['form_bg_color'] ) ?: $defaults['form_bg_color'];
    $out['form_bg_opacity'] = isset( $input['form_bg_opacity'] ) ? max( 0, min( 1, floatval( $input['form_bg_opacity'] ) ) ) : $defaults['form_bg_opacity'];
    $out['blur_amount']    = isset( $input['blur_amount'] ) ? absint( $input['blur_amount'] ) : $defaults['blur_amount'];
    $out['border_radius']  = isset( $input['border_radius'] ) ? absint( $input['border_radius'] ) : $defaults['border_radius'];

    $out['primary_color']  = sanitize_hex_color( $input['primary_color'] ?? $defaults['primary_color'] ) ?: $defaults['primary_color'];
    $out['button_color_1'] = sanitize_hex_color( $input['button_color_1'] ?? $defaults['button_color_1'] ) ?: $defaults['button_color_1'];
    $out['button_color_2'] = sanitize_hex_color( $input['button_color_2'] ?? $defaults['button_color_2'] ) ?: $defaults['button_color_2'];
    $out['text_color']     = sanitize_hex_color( $input['text_color'] ?? $defaults['text_color'] ) ?: $defaults['text_color'];
    $out['label_color']    = sanitize_hex_color( $input['label_color'] ?? $defaults['label_color'] ) ?: $defaults['label_color'];

    $allowed_fonts = array( 'vazirmatn', 'system' );
    $out['font_choice'] = in_array( $input['font_choice'] ?? '', $allowed_fonts, true ) ? $input['font_choice'] : $defaults['font_choice'];

    $out['shadow_enabled']  = ! empty( $input['shadow_enabled'] ) ? 1 : 0;
    $out['floating_shapes'] = ! empty( $input['floating_shapes'] ) ? 1 : 0;

    $out['logo_image'] = isset( $input['logo_image'] ) ? esc_url_raw( $input['logo_image'] ) : $defaults['logo_image'];
    $out['logo_width'] = isset( $input['logo_width'] ) ? absint( $input['logo_width'] ) : $defaults['logo_width'];
    $out['logo_link']  = isset( $input['logo_link'] ) ? esc_url_raw( $input['logo_link'] ) : $defaults['logo_link'];

    $out['welcome_text'] = isset( $input['welcome_text'] ) ? sanitize_text_field( $input['welcome_text'] ) : $defaults['welcome_text'];

    $out['hide_back_to_blog']      = ! empty( $input['hide_back_to_blog'] ) ? 1 : 0;
    $out['hide_language_switcher'] = ! empty( $input['hide_language_switcher'] ) ? 1 : 0;

    return $out;
}

add_filter( 'login_message', 'bld_login_message' );
function bld_login_message( $message ) {
    $o = bld_get_options();
    if ( empty( $o['welcome_text'] ) ) {
        return $message;
    }
    $welcome = '<p id="bld-welcome-text">' . esc_html( $o['welcome_text'] ) . '</p>';
    return $welcome . $message;
}

add_filter( 'login_headerurl', 'bld_login_logo_url' );
function bld_login_logo_url() {
    $o = bld_get_options();
    return ! empty( $o['logo_link'] ) ? esc_url( $o['logo_link'] ) : home_url( '/' );
}

function bld_login_logo_text() {
    return get_bloginfo( 'name' );
}

 * 6) توابع کمکی
 * ============================================================ */
function bld_hex_to_rgb( $hex ) {
    $hex = ltrim( $hex, '#' );
    if ( strlen( $hex ) === 3 ) {
        $hex = $hex[0] . $hex[0] . $hex[1] . $hex[1] . $hex[2] . $hex[2];
    }
    if ( strlen( $hex ) !== 6 ) {
        return '255,255,255';
    }
    $r = hexdec( substr( $hex, 0, 2 ) );
    $g = hexdec( substr( $hex, 2, 2 ) );
    $b = hexdec( substr( $hex, 4, 2 ) );
    return "$r,$g,$b";
}
