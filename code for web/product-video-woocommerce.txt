

if ( ! defined( 'ABSPATH' ) ) {
	exit; 
}

add_action( 'plugins_loaded', function () {

	if ( ! class_exists( 'WooCommerce' ) ) {
		return; 
	}

	if ( class_exists( 'PVW_Product_Video' ) ) {
		return; 
	}

	class PVW_Product_Video {

		const META_KEY     = '_pvw_product_video_id';
		const NONCE_ACTION = 'pvw_save_product_video';
		const NONCE_NAME   = 'pvw_product_video_nonce';

		public function __construct() {
			add_action( 'add_meta_boxes', array( $this, 'add_metabox' ) );
			add_action( 'admin_enqueue_scripts', array( $this, 'enqueue_admin_assets' ) );
			add_action( 'save_post_product', array( $this, 'save_video_field' ) );
			add_filter( 'woocommerce_single_product_image_thumbnail_html', array( $this, 'replace_main_image_with_video' ), 20, 2 );
			add_action( 'wp_head', array( $this, 'print_front_css' ) );
			add_action( 'wp_enqueue_scripts', array( $this, 'enqueue_front_assets' ) );
		}


		public function add_metabox() {
			add_meta_box(
				'pvw_product_video_box',
				__( '🎬 ویدیوی محصول (جایگزین تصویر اصلی)', 'pvw' ),
				array( $this, 'render_metabox' ),
				'product',
				'side',
				'high'
			);
		}

		public function render_metabox( $post ) {
			wp_nonce_field( self::NONCE_ACTION, self::NONCE_NAME );
			$video_id  = get_post_meta( $post->ID, self::META_KEY, true );
			$video_url = $video_id ? wp_get_attachment_url( $video_id ) : '';
			?>
			<div id="pvw-video-box" style="text-align:right;direction:rtl;">
				<div id="pvw-video-preview" style="margin-bottom:10px;">
					<?php if ( $video_url ) : ?>
						<video src="<?php echo esc_url( $video_url ); ?>" style="width:100%;border-radius:6px;display:block;" muted controls></video>
					<?php else : ?>
						<p style="color:#777;font-size:12px;margin:0 0 8px;"><?php esc_html_e( 'هنوز ویدیویی انتخاب نشده است.', 'pvw' ); ?></p>
					<?php endif; ?>
				</div>

				<input type="hidden" id="pvw_video_id" name="pvw_video_id" value="<?php echo esc_attr( $video_id ); ?>">

				<button type="button" class="button button-primary" id="pvw-select-video" style="width:100%;margin-bottom:6px;">
					<?php echo $video_id ? esc_html__( 'تغییر ویدیو', 'pvw' ) : esc_html__( 'انتخاب ویدیو', 'pvw' ); ?>
				</button>

				<button type="button" class="button" id="pvw-remove-video" style="width:100%;<?php echo $video_id ? '' : 'display:none;'; ?>">
					<?php esc_html_e( 'حذف ویدیو', 'pvw' ); ?>
				</button>

				<p style="font-size:11px;color:#888;margin-top:10px;line-height:1.6;">
					<?php esc_html_e( 'این ویدیو به‌جای تصویر شاخص محصول، در صفحه محصول نمایش داده می‌شود و کاملاً جایگزین تصویر اصلی می‌گردد (نه گالری).', 'pvw' ); ?>
				</p>
			</div>
			<?php
		}

		public function enqueue_admin_assets( $hook ) {
			if ( ! in_array( $hook, array( 'post.php', 'post-new.php' ), true ) ) {
				return;
			}

			global $post;
			if ( ! isset( $post ) || 'product' !== $post->post_type ) {
				return;
			}

			wp_enqueue_media();

			$select_label = esc_js( __( 'انتخاب ویدیوی محصول', 'pvw' ) );
			$button_label = esc_js( __( 'استفاده از این ویدیو', 'pvw' ) );
			$change_label = esc_js( __( 'تغییر ویدیو', 'pvw' ) );
			$empty_label  = esc_js( __( 'هنوز ویدیویی انتخاب نشده است.', 'pvw' ) );

			$js = "
			jQuery(function($){
				var pvwFrame;
				$('#pvw-select-video').on('click', function(e){
					e.preventDefault();
					if ( pvwFrame ) { pvwFrame.open(); return; }
					pvwFrame = wp.media({
						title: '{$select_label}',
						library: { type: 'video' },
						multiple: false,
						button: { text: '{$button_label}' }
					});
					pvwFrame.on('select', function(){
						var attachment = pvwFrame.state().get('selection').first().toJSON();
						$('#pvw_video_id').val(attachment.id);
						$('#pvw-video-preview').html('<video src=\"' + attachment.url + '\" style=\"width:100%;border-radius:6px;display:block;\" muted controls></video>');
						$('#pvw-remove-video').show();
						$('#pvw-select-video').text('{$change_label}');
					});
					pvwFrame.open();
				});

				$('#pvw-remove-video').on('click', function(e){
					e.preventDefault();
					$('#pvw_video_id').val('');
					$('#pvw-video-preview').html('<p style=\"color:#777;font-size:12px;margin:0 0 8px;\">{$empty_label}</p>');
					$(this).hide();
				});
			});
			";

			wp_add_inline_script( 'jquery', $js );
		}


		public function save_video_field( $post_id ) {

			if ( defined( 'DOING_AUTOSAVE' ) && DOING_AUTOSAVE ) {
				return;
			}

			if ( ! isset( $_POST[ self::NONCE_NAME ] ) ||
				! wp_verify_nonce( sanitize_text_field( wp_unslash( $_POST[ self::NONCE_NAME ] ) ), self::NONCE_ACTION ) ) {
				return;
			}

			if ( ! current_user_can( 'edit_product', $post_id ) ) {
				return;
			}

			if ( ! isset( $_POST['pvw_video_id'] ) ) {
				return;
			}

			$video_id = absint( $_POST['pvw_video_id'] );

			if ( $video_id > 0 ) {
				$mime = get_post_mime_type( $video_id );
				if ( $mime && strpos( $mime, 'video/' ) === 0 ) {
					update_post_meta( $post_id, self::META_KEY, $video_id );
				}
			} else {
				delete_post_meta( $post_id, self::META_KEY );
			}
		}


		public function replace_main_image_with_video( $html, $attachment_id ) {
			global $product;

			if ( ! $product instanceof WC_Product ) {
				return $html;
			}

			$main_image_id = $product->get_image_id();
			if ( ! $main_image_id || (int) $attachment_id !== (int) $main_image_id ) {
				return $html;
			}

			$video_id = get_post_meta( $product->get_id(), self::META_KEY, true );
			if ( ! $video_id ) {
				return $html;
			}

			$video_url = wp_get_attachment_url( $video_id );
			if ( ! $video_url ) {
				return $html;
			}

			$mime = get_post_mime_type( $video_id );
			if ( ! $mime || strpos( $mime, 'video/' ) !== 0 ) {
				return $html;
			}

			$alt = get_post_meta( $main_image_id, '_wp_attachment_image_alt', true );
			$alt = $alt ? $alt : get_the_title( $product->get_id() );

		
			$video_tag = sprintf(
				'<video class="wp-post-image pvw-product-video" autoplay muted loop playsinline disablepictureinpicture preload="auto" aria-label="%1$s"><source src="%2$s" type="%3$s"></video>',
				esc_attr( $alt ),
				esc_url( $video_url ),
				esc_attr( $mime )
			);


			$new_html = preg_replace( '/<img\b[^>]*>/i', $video_tag, $html, 1 );

			return $new_html ? $new_html : $html;
		}

		public function print_front_css() {
			if ( ! function_exists( 'is_product' ) || ! is_product() ) {
				return;
			}
			?>
			<style>
				/* اجبار کامل ویدیو به گرفتن دقیقاً همان عرض و ارتفاعی که تصویر می‌گرفت */
				.woocommerce-product-gallery__image video.pvw-product-video,
				.pvw-product-video {
					display: block !important;
					width: 100% !important;
					height: auto !important;
					max-width: 100% !important;
					vertical-align: bottom !important;
					object-fit: cover;
					background: transparent;
					border: 0;
					box-shadow: none;
				}
				.woocommerce-product-gallery__image:has(video) {
					line-height: 0;
				}
			</style>
			<?php
		}


		public function enqueue_front_assets() {
			if ( ! function_exists( 'is_product' ) || ! is_product() ) {
				return;
			}

			$js = "
			(function(){
				function pvwPlayVisible(){
					document.querySelectorAll('.pvw-product-video').forEach(function(v){
						var wrap = v.closest('.woocommerce-product-gallery__image');
						var el = wrap || v;
						var style = window.getComputedStyle(el);
						if ( style.display !== 'none' ) {
							v.muted = true;
							var p = v.play();
							if ( p && typeof p.catch === 'function' ) { p.catch(function(){}); }
						}
					});
				}

				document.addEventListener('DOMContentLoaded', function(){
					pvwPlayVisible();

					if ( window.jQuery ) {
						jQuery(document.body).on( 'woocommerce_gallery_init_zoom wc-product-gallery-after-init init.flexslider after.flexslider', function(){
							setTimeout( pvwPlayVisible, 50 );
						});
						jQuery(document).on( 'click', '.flex-control-nav a, .woocommerce-product-gallery__image a', function(){
							setTimeout( pvwPlayVisible, 100 );
						});
					}

					document.addEventListener('click', function(e){
						var v = e.target.closest ? e.target.closest('.pvw-product-video') : null;
						if ( v ) {
							e.preventDefault();
							e.stopPropagation();
						}
					}, true);

					var gallery = document.querySelector('.woocommerce-product-gallery');
					if ( gallery && window.MutationObserver ) {
						new MutationObserver(function(){ pvwPlayVisible(); }).observe( gallery, { attributes:true, subtree:true, attributeFilter:['style','class'] } );
					}
				});
			})();
			";

			wp_register_script( 'pvw-front', false, array(), '1.0', true );
			wp_enqueue_script( 'pvw-front' );
			wp_add_inline_script( 'pvw-front', $js );
		}
	}

	new PVW_Product_Video();

} );
