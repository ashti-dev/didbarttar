function irwds_default_settings() {
	return array(
		'enabled'            => 1,
		'hour'               => 11,
		'minute'             => 0,
		'icon'               => '🚀',
		'text_hour_tpl'      => 'اگر تا {time} دیگر خرید کنید، سفارشتان همین امروز ارسال می‌شود',
		'text_minute_tpl'    => 'اگر تا {time} دیگر خرید کنید، سفارشتان همین امروز ارسال می‌شود',
		'expired_behavior'   => 'hide', // hide | message
		'expired_text'       => 'زمان ارسال فوری امروز به پایان رسید، فردا دوباره فعال می‌شود ',
		'link_url'           => '',
		'show_close'         => 1,
		'color_start'        => '#ff4e50',
		'color_end'          => '#f9004d',
		'text_color'         => '#ffffff',
		'timer_bg'           => 'rgba(255,255,255,0.18)',
	);
}

function irwds_get_settings() {
	$saved = get_option( IRWDS_OPTION_KEY, array() );
	if ( ! is_array( $saved ) ) {
		$saved = array();
	}
	return array_merge( irwds_default_settings(), $saved );
}

function irwds_sanitize_settings( $input ) {
	$defaults = irwds_default_settings();
	$out      = array();

	$out['enabled']     = isset( $input['enabled'] ) ? 1 : 0;
	$out['hour']        = max( 0, min( 23, absint( $input['hour'] ?? $defaults['hour'] ) ) );
	$out['minute']      = max( 0, min( 59, absint( $input['minute'] ?? $defaults['minute'] ) ) );
	$out['icon']        = sanitize_text_field( $input['icon'] ?? $defaults['icon'] );
	$out['text_hour_tpl']   = sanitize_text_field( $input['text_hour_tpl'] ?? $defaults['text_hour_tpl'] );
	$out['text_minute_tpl'] = sanitize_text_field( $input['text_minute_tpl'] ?? $defaults['text_minute_tpl'] );
	$out['expired_behavior'] = in_array( $input['expired_behavior'] ?? '', array( 'hide', 'message' ), true )
		? $input['expired_behavior'] : $defaults['expired_behavior'];
	$out['expired_text'] = sanitize_text_field( $input['expired_text'] ?? $defaults['expired_text'] );
	$out['link_url']     = ! empty( $input['link_url'] ) ? esc_url_raw( $input['link_url'] ) : '';
	$out['show_close']   = isset( $input['show_close'] ) ? 1 : 0;

	foreach ( array( 'color_start', 'color_end', 'text_color' ) as $c ) {
		$val         = sanitize_text_field( $input[ $c ] ?? $defaults[ $c ] );
		$out[ $c ]   = sanitize_hex_color( $val ) ? $val : $defaults[ $c ];
	}
	$out['timer_bg'] = sanitize_text_field( $input['timer_bg'] ?? $defaults['timer_bg'] );

	return $out;
}

function irwds_calculate_target() {
	$s  = irwds_get_settings();
	$tz = new DateTimeZone( 'Asia/Tehran' );

	$now    = new DateTime( 'now', $tz );
	$target = new DateTime( 'now', $tz );
	$target->setTime( (int) $s['hour'], (int) $s['minute'], 0 );

	if ( $target <= $now ) {
		$target->modify( '+1 day' );
	}

	return array(
		'now_ts'    => $now->getTimestamp(),
		'target_ts' => $target->getTimestamp(),
	);
}


add_action( 'wp_head', 'irwds_enqueue_font' );
function irwds_enqueue_font() {
	if ( is_admin() ) {
		return;
	}
	$s = irwds_get_settings();
	if ( empty( $s['enabled'] ) ) {
		return;
	}
	echo '<link rel="preconnect" href="https://fonts.googleapis.com">';
	echo '<link rel="preconnect" href="https://fonts.gstatic.com" crossorigin>';
	echo '<link href="https://fonts.googleapis.com/css2?family=Vazirmatn:wght@500;700;800&display=swap" rel="stylesheet">';
}

	(function () {

		var serverMs   = <?php echo (int) $server_ms; ?>;
		var clientMs   = Date.now();
		var offset     = serverMs - clientMs;

		var minuteTpl  = <?php echo wp_json_encode( $s['text_minute_tpl'] ); ?>;

			var str = String(num);
			var out = '';
			for (var i = 0; i < str.length; i++) {

			return (n < 10 ? '0' + n : '' + n);
		}

		function adjustBodyOffset() {

				return;

			document.body.style.paddingTop = bar.offsetHeight + 'px';

			if (diff <= 0) {
				if (expiredBehavior === 'hide') {
					if (bar) { bar.style.display = 'none'; }
					document.body.style.paddingTop = '';

					return;

					if (timerEl) { timerEl.style.display = 'none'; }
					if (textEl) { textEl.textContent = expiredText; }

					setTimeout(function(){ location.reload(); }, 5 * 60 * 1000);

			if (mmEl) mmEl.textContent = toFa(pad2(minutes));

				timeLabel = toFa(minutes) + ' دقیقه';
				tpl = minuteTpl;
			} else {
				timeLabel = toFa(seconds) + ' ثانیه';
				tpl = minuteTpl;
			}
			if (textEl) {
				textEl.textContent = tpl.replace('{time}', timeLabel);
			}

			adjustBodyOffset();
		}

		window.addEventListener('resize', adjustBodyOffset);
		window.addEventListener('load', adjustBodyOffset);
	})();
	</script>

}
