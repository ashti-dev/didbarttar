if ( ! class_exists( 'WPS_Order_Tracking_System' ) ) :

final class WPS_Order_Tracking_System {

	const META_CARRIER   = '_wps_tracking_carrier';
	const META_CODE       = '_wps_tracking_code';
	const OPTION_KEY       = 'wps_tracking_settings';
	const NONCE_ACTION     = 'wps_tracking_save_nonce';

	private static $instance = null;

	public static function instance() {
		if ( null === self::$instance ) {
			self::$instance = new self();
		}
		return self::$instance;
	}

	private function __construct() {
		// تنظیمات پیش‌فرض
		add_action( 'init', array( $this, 'maybe_set_defaults' ) );

		// باکس متا در صفحه سفارش (سازگار با HPOS و روش قدیمی)
		add_action( 'add_meta_boxes', array( $this, 'register_meta_box' ) );
		add_action( 'woocommerce_process_shop_order_meta', array( $this, 'save_meta_box' ), 10, 1 );
		add_action( 'save_post_shop_order', array( $this, 'save_meta_box' ), 10, 1 );

		// نمایش در پنل مشتری (My Account -> View Order)
		add_action( 'woocommerce_order_details_after_order_table', array( $this, 'render_customer_tracking_box' ), 5 );

		// استایل‌ها
		add_action( 'wp_enqueue_scripts', array( $this, 'enqueue_frontend_styles' ) );
		add_action( 'admin_enqueue_scripts', array( $this, 'enqueue_admin_styles' ) );

		// منوی تنظیمات
		add_action( 'admin_menu', array( $this, 'register_settings_page' ) );
		add_action( 'admin_init', array( $this, 'register_settings' ) );
	}

	/** تنظیمات پیش‌فرض در اولین اجرا */
	public function maybe_set_defaults() {
		if ( false === get_option( self::OPTION_KEY ) ) {
			update_option( self::OPTION_KEY, $this->default_settings() );
		}
	}

	private function default_settings() {
		return array(
			'title'          => 'پیگیری مرسوله شما',
			'subtitle'       => 'وضعیت ارسال بسته خود را به صورت آنی مشاهده کنید',
			// پالت جدید: سفید/روشن و ساده، با یک رنگ اصلی (accent) برای جلب توجه بدون شلوغی
			'color_primary'  => '#ffffff',
			'color_secondary'=> '#f7f8fc',
			'color_accent'   => '#4f46e5',
			'post_url'       => 'https://tracking.post.ir/',
			'chapar_url'     => 'https://www.chaparnet.com/track',
			'tipax_url'      => 'https://tipaxco.com/tracking',
			'btn_text'       => 'پیگیری مرسوله',
		);
	}

	private function get_settings() {
		$defaults = $this->default_settings();
		$saved    = get_option( self::OPTION_KEY, array() );
		return wp_parse_args( $saved, $defaults );
	}

	/** لیست شرکت‌های حمل */
	private function carriers() {

			'post'   => 'پست ایران',
			'chapar' => 'چاپار',

	public function register_meta_box() {
		$screen = class_exists( '\Automattic\WooCommerce\Utilities\OrderUtil' )
			&& \Automattic\WooCommerce\Utilities\OrderUtil::custom_orders_table_usage_is_enabled()
			? wc_get_page_screen_id( 'shop-order' )
			: 'shop_order';

			array( $this, 'meta_box_html' ),
			$screen,
			'side',
			'high'
		);
	}

	public function meta_box_html( $post_or_order ) {
		$order = ( $post_or_order instanceof WP_Post ) ? wc_get_order( $post_or_order->ID ) : $post_or_order;
		if ( ! $order instanceof WC_Order ) {
			return;
		}

		$carrier = $order->get_meta( self::META_CARRIER );
		$code    = $order->get_meta( self::META_CODE );
		$carriers = $this->carriers();

		wp_nonce_field( self::NONCE_ACTION, 'wps_tracking_nonce' );
		?>
		<p>
			<label for="wps_tracking_carrier" style="font-weight:600;display:block;margin-bottom:4px;">شرکت پستی/باربری:</label>
			<select name="wps_tracking_carrier" id="wps_tracking_carrier" style="width:100%;">
				<option value=""><?php esc_html_e( '— انتخاب کنید —' ); ?></option>
				<?php foreach ( $carriers as $key => $label ) : ?>
					<option value="<?php echo esc_attr( $key ); ?>" <?php selected( $carrier, $key ); ?>>
						<?php echo esc_html( $label ); ?>
					</option>
				<?php endforeach; ?>
			</select>
		</p>
		<p>
			<label for="wps_tracking_code" style="font-weight:600;display:block;margin-bottom:4px;">کد رهگیری مرسوله:</label>
			<input type="text" name="wps_tracking_code" id="wps_tracking_code"
				value="<?php echo esc_attr( $code ); ?>" style="width:100%;" placeholder="مثال: 1234567890" />
		</p>
		<p style="color:#666;font-size:12px;">
			پس از ذخیره سفارش، این کد به همراه لینک پیگیری زنده در پنل کاربری مشتری نمایش داده می‌شود.
		</p>
		<?php
	}

	public function save_meta_box( $order_id ) {
		if ( ! isset( $_POST['wps_tracking_nonce'] ) ||
			! wp_verify_nonce( sanitize_text_field( wp_unslash( $_POST['wps_tracking_nonce'] ) ), self::NONCE_ACTION ) ) {
			return;
		}

		if ( ! current_user_can( 'edit_shop_orders' ) && ! current_user_can( 'manage_woocommerce' ) ) {
			return;
		}

		$order = wc_get_order( $order_id );
		if ( ! $order instanceof WC_Order ) {
			return;
		}

		$carriers = array_keys( $this->carriers() );

		if ( isset( $_POST['wps_tracking_carrier'] ) ) {
			$carrier = sanitize_text_field( wp_unslash( $_POST['wps_tracking_carrier'] ) );
			if ( '' === $carrier || in_array( $carrier, $carriers, true ) ) {
				$order->update_meta_data( self::META_CARRIER, $carrier );
			}
		}

		if ( isset( $_POST['wps_tracking_code'] ) ) {
			$code = sanitize_text_field( wp_unslash( $_POST['wps_tracking_code'] ) );
			$order->update_meta_data( self::META_CODE, $code );
		}

		$order->save();
	}

	public function render_customer_tracking_box( $order ) {
		if ( ! $order instanceof WC_Order ) {
			return;
		}

		$carrier = $order->get_meta( self::META_CARRIER );
		$code    = $order->get_meta( self::META_CODE );

		if ( empty( $carrier ) || empty( $code ) ) {
			return; // اگر کد رهگیری ثبت نشده باشد چیزی نمایش داده نمی‌شود
		}

		$settings = $this->get_settings();
		$carriers = $this->carriers();
		$carrier_label = isset( $carriers[ $carrier ] ) ? $carriers[ $carrier ] : '';

		$track_url = '';
		switch ( $carrier ) {
			case 'post':
				$track_url = $settings['post_url'];
				break;
			case 'chapar':
				$track_url = $settings['chapar_url'];
				break;
			case 'tipax':
				$track_url = $settings['tipax_url'];
				break;
		}

		$icon_svg     = $this->get_carrier_icon( $carrier );
		$carrier_hex  = $this->get_carrier_accent( $carrier );
		?>
		<div class="wps-tracking-card" dir="rtl" style="--chip-color: <?php echo esc_attr( $carrier_hex ); ?>;">
			<div class="wps-tracking-top-strip"></div>

			<div class="wps-tracking-inner">

				<div class="wps-header-row">
					<span class="wps-header-icon"><?php echo $icon_svg; // phpcs:ignore - SVG داخلی امن ?></span>
					<div class="wps-header-text">
						<div class="wps-eyebrow"><?php echo esc_html( $settings['title'] ); ?></div>
						<p class="wps-tracking-sub"><?php echo esc_html( $settings['subtitle'] ); ?></p>
					</div>
				</div>

				<div class="wps-top-row">
					<div class="wps-carrier-chip">
						<span class="wps-carrier-name"><?php echo esc_html( $carrier_label ); ?></span>
					</div>

					<div class="wps-code-capsule">
						<div class="wps-code-text-wrap">
							<span class="wps-code-label">کد رهگیری</span>
							<span class="wps-code-value" id="wps-code-text"><?php echo esc_html( $code ); ?></span>
						</div>
						<button type="button" class="wps-copy-btn" data-copy-target="wps-code-text" aria-label="کپی کد رهگیری">
							<svg class="wps-copy-icon" width="16" height="16" viewBox="0 0 24 24" fill="none" xmlns="http://www.w3.org/2000/svg">
								<rect x="8" y="8" width="12" height="12" rx="2" stroke="currentColor" stroke-width="1.8"/>
								<path d="M4 16V6C4 4.89543 4.89543 4 6 4H16" stroke="currentColor" stroke-width="1.8" stroke-linecap="round"/>
							</svg>
							<span class="wps-copy-txt">کپی</span>
						</button>
					</div>
				</div>

				<?php if ( $track_url ) : ?>
					<a class="wps-tracking-cta" href="<?php echo esc_url( $track_url ); ?>" target="_blank" rel="noopener noreferrer">
						<span><?php echo esc_html( $settings['btn_text'] ); ?></span>
						<svg width="18" height="18" viewBox="0 0 24 24" fill="none" xmlns="http://www.w3.org/2000/svg">
							<path d="M14 5L21 12M21 12L14 19M21 12H3" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"/>
						</svg>
					</a>
				<?php endif; ?>
			</div>
		</div>
		<?php
	}

	private function get_carrier_accent( $carrier ) {
		$map = array(
			'post'   => '#60a5fa', // آبی روشن
			'chapar' => '#fbbf24', // کهربایی
			'tipax'  => '#34d399', // سبز نعنایی
		);
		return isset( $map[ $carrier ] ) ? $map[ $carrier ] : '#94a3b8';
	}

	public function enqueue_frontend_styles() {
		if ( ! function_exists( 'is_account_page' ) || ! is_account_page() ) {
			return;
		}

		$settings = $this->get_settings();
		$css = $this->build_css( $settings['color_primary'], $settings['color_secondary'], $settings['color_accent'] );

		wp_register_style( 'wps-tracking-style', false );
		wp_enqueue_style( 'wps-tracking-style' );
		wp_add_inline_style( 'wps-tracking-style', $css );

		$js = "document.addEventListener('click', function(e){
			var btn = e.target.closest('.wps-copy-btn');
			if(!btn) return;
			var targetId = btn.getAttribute('data-copy-target');
			var el = document.getElementById(targetId);
			if(!el) return;
			var text = el.textContent.trim();
			var label = btn.querySelector('.wps-copy-txt');
			if(!navigator.clipboard){ return; }
			navigator.clipboard.writeText(text).then(function(){
				var original = label ? label.textContent : '';
				if(label){ label.textContent = 'کپی شد'; }
				btn.classList.add('wps-copied');
				setTimeout(function(){
					if(label){ label.textContent = original; }
					btn.classList.remove('wps-copied');
				}, 1800);
			});
		});";
		wp_register_script( 'wps-tracking-script', false, array(), '1.0', true );
		wp_enqueue_script( 'wps-tracking-script' );
		wp_add_inline_script( 'wps-tracking-script', $js );
	}

	public function register_settings_page() {
		add_submenu_page(
			'woocommerce',
			'تنظیمات پیگیری مرسوله',

	public function register_settings() {
		register_setting( 'wps_tracking_settings_group', self::OPTION_KEY, array( $this, 'sanitize_settings' ) );
	}

	public function sanitize_settings( $input ) {
		$clean = array();
		$clean['title']           = isset( $input['title'] ) ? sanitize_text_field( $input['title'] ) : '';
		$clean['subtitle']        = isset( $input['subtitle'] ) ? sanitize_text_field( $input['subtitle'] ) : '';
		$clean['btn_text']        = isset( $input['btn_text'] ) ? sanitize_text_field( $input['btn_text'] ) : '';
		$clean['color_primary']   = isset( $input['color_primary'] ) && sanitize_hex_color( $input['color_primary'] ) ? $input['color_primary'] : '#ffffff';
		$clean['color_secondary'] = isset( $input['color_secondary'] ) && sanitize_hex_color( $input['color_secondary'] ) ? $input['color_secondary'] : '#f7f8fc';
		$clean['color_accent']    = isset( $input['color_accent'] ) && sanitize_hex_color( $input['color_accent'] ) ? $input['color_accent'] : '#4f46e5';
		$clean['post_url']        = isset( $input['post_url'] ) ? esc_url_raw( $input['post_url'] ) : '';
		$clean['chapar_url']      = isset( $input['chapar_url'] ) ? esc_url_raw( $input['chapar_url'] ) : '';
		$clean['tipax_url']       = isset( $input['tipax_url'] ) ? esc_url_raw( $input['tipax_url'] ) : '';
		return $clean;
	}
