function clx_bar_default_settings() {
	return array(
		'enabled'          => 1,
		'start_datetime'   => date( 'Y-m-d\TH:i' ),
		'end_datetime'     => date( 'Y-m-d\TH:i', strtotime( '+3 days' ) ),
		'message'          => '🔥 جشنواره ویژه تابستانه فقط تا پایان زمان!',
		'coupon_text'      => 'کد: SUMMER26',
		'button_text'      => 'مشاهده تخفیف‌ها',
		'button_url'       => '',
		'expire_mode'      => 'message', // hide | message
		'expire_message'   => 'مهلت این پیشنهاد به پایان رسید.',
		'closable'         => 1,
		'persian_digits'   => 1,
		'bar_size'         => 'normal', // normal | compact
		'bg_color_1'       => '#0B1F26',
		'bg_color_2'       => '#0EA5A4',
		'text_color'       => '#FFF8EC',
		'accent_color'     => '#D4AF37',
		'digit_bg_color'   => 'rgba(255,255,255,.12)',
	);
}

function clx_bar_get_settings() {
	$saved = get_option( CLX_BAR_OPT, array() );
	return wp_parse_args( $saved, clx_bar_default_settings() );
}

add_action( 'admin_post_clx_bar_save_settings', 'clx_bar_save_settings' );
function clx_bar_save_settings() {
	if ( ! current_user_can( 'manage_options' ) ) { wp_die( 'دسترسی ندارید.' ); }
	check_admin_referer( CLX_BAR_NONCE );

	$settings = array(
		'enabled'        => isset( $_POST['enabled'] ) ? 1 : 0,
		'start_datetime' => sanitize_text_field( $_POST['start_datetime'] ?? '' ),
		'end_datetime'   => sanitize_text_field( $_POST['end_datetime'] ?? '' ),
		'message'        => sanitize_text_field( $_POST['message'] ?? '' ),
		'coupon_text'    => sanitize_text_field( $_POST['coupon_text'] ?? '' ),
		'button_text'    => sanitize_text_field( $_POST['button_text'] ?? '' ),
		'button_url'     => esc_url_raw( $_POST['button_url'] ?? '' ),
		'expire_mode'    => sanitize_text_field( $_POST['expire_mode'] ?? 'message' ),
		'expire_message' => sanitize_text_field( $_POST['expire_message'] ?? '' ),
		'closable'       => isset( $_POST['closable'] ) ? 1 : 0,
		'persian_digits' => isset( $_POST['persian_digits'] ) ? 1 : 0,
		'bar_size'       => sanitize_text_field( $_POST['bar_size'] ?? 'normal' ),
		'bg_color_1'     => sanitize_hex_color( $_POST['bg_color_1'] ?? '' ) ?: '#0B1F26',
		'bg_color_2'     => sanitize_hex_color( $_POST['bg_color_2'] ?? '' ) ?: '#0EA5A4',
		'text_color'     => sanitize_hex_color( $_POST['text_color'] ?? '' ) ?: '#FFF8EC',
		'accent_color'   => sanitize_hex_color( $_POST['accent_color'] ?? '' ) ?: '#D4AF37',
		'digit_bg_color' => sanitize_text_field( $_POST['digit_bg_color'] ?? 'rgba(255,255,255,.12)' ),
	);

	update_option( CLX_BAR_OPT, $settings );
	wp_safe_redirect( admin_url( 'admin.php?page=clx-discount-bar&updated=1' ) );
	exit;
}

			<?php endif; ?>

		<div id="clx-bar-progress-track"><div id="clx-bar-progress-fill"></div></div>

	<script>
	(function(){
		var startTs   = <?php echo (int) $start_ts; ?> * 1000;
		var endTs     = <?php echo (int) $end_ts; ?> * 1000;
		var expireMode    = <?php echo wp_json_encode( $s['expire_mode'] ); ?>;
		var expireMessage = <?php echo wp_json_encode( $s['expire_message'] ); ?>;
		var usePersianDigits = <?php echo $s['persian_digits'] ? 'true' : 'false'; ?>;

		var wrap = document.getElementById('clx-bar-wrap');
		var closeBtn = document.getElementById('clx-bar-close');

		if (sessionStorage.getItem('clx_bar_closed')) { return; }

		function toPersianDigits(str){
			if (!usePersianDigits) return str;
			var fa = ['۰','۱','۲','۳','۴','۵','۶','۷','۸','۹'];
			return String(str).replace(/[0-9]/g, function(d){ return fa[d]; });
		}
		function pad(n){ return (n<10?'0':'')+n; }

		function adjustBodySpacing(){
			var h = wrap.offsetHeight;
			document.body.style.marginTop = h + 'px';
		}

		function renderExpired(){
			if (expireMode === 'hide') {
				wrap.style.display = 'none';
				document.body.style.marginTop = '';
				return;
			}
			document.getElementById('clx-bar-msg').textContent = expireMessage;
			document.getElementById('clx-bar-timer').style.display = 'none';
			var coupon = document.getElementById('clx-bar-coupon');
			if (coupon) coupon.style.display = 'none';
			document.getElementById('clx-bar-progress-fill').style.width = '100%';
		}

		function tick(){
			var now = Date.now();
			if (now >= endTs) {
				clearInterval(timerHandle);
				renderExpired();
				return;
			}
			var remaining = endTs - now;
			var d = Math.floor(remaining / 86400000);
			var h = Math.floor((remaining % 86400000) / 3600000);
			var m = Math.floor((remaining % 3600000) / 60000);
			var s = Math.floor((remaining % 60000) / 1000);

			document.querySelector('[data-unit="d"]').textContent = toPersianDigits(pad(d));
			document.querySelector('[data-unit="h"]').textContent = toPersianDigits(pad(h));
			document.querySelector('[data-unit="m"]').textContent = toPersianDigits(pad(m));
			document.querySelector('[data-unit="s"]').textContent = toPersianDigits(pad(s));

			var total = endTs - startTs;
			var elapsed = now - startTs;
			var pct = total > 0 ? Math.min(100, Math.max(0, (elapsed/total)*100)) : 100;
			document.getElementById('clx-bar-progress-fill').style.width = pct + '%';
		}

		var timerHandle = setInterval(tick, 1000);
		tick();

		requestAnimationFrame(function(){
			wrap.classList.add('clx-bar-show');
			adjustBodySpacing();
		});
		window.addEventListener('resize', adjustBodySpacing);

		if (closeBtn) {
			closeBtn.addEventListener('click', function(){
				wrap.classList.remove('clx-bar-show');
				document.body.style.marginTop = '';
				sessionStorage.setItem('clx_bar_closed','1');
			});
		}

		var copyBtn = document.getElementById('clx-bar-copy-btn');
		if (copyBtn) {
			copyBtn.addEventListener('click', function(){
				var raw = document.getElementById('clx-bar-coupon-text').textContent.trim();
				var code = raw.replace(/^(کد|code)\s*[:：]\s*/i, '');
				var doCopyFeedback = function(){
					var original = copyBtn.innerHTML;
					copyBtn.innerHTML = '✓ کپی شد';
					copyBtn.classList.add('clx-copied');
					setTimeout(function(){
						copyBtn.innerHTML = original;
						copyBtn.classList.remove('clx-copied');
					}, 1500);
				};
				if (navigator.clipboard && navigator.clipboard.writeText) {
					navigator.clipboard.writeText(code).then(doCopyFeedback).catch(function(){
						fallbackCopy(code, doCopyFeedback);
					});
				} else {
					fallbackCopy(code, doCopyFeedback);
				}
			});
		}

		function fallbackCopy(text, cb){
			var ta = document.createElement('textarea');
			ta.value = text;
			ta.style.position = 'fixed';
			ta.style.opacity = '0';
			document.body.appendChild(ta);
			ta.select();
			try { document.execCommand('copy'); } catch(e){}
			document.body.removeChild(ta);
			cb();
		}
	})();
	</script>
