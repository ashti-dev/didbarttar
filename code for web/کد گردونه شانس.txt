function clx_ws_get_settings() {
	$saved = get_option( CLX_WS_OPT_SETTINGS, array() );
	return wp_parse_args( $saved, clx_ws_default_settings() );
}

function clx_ws_get_items() {
	$items = get_option( CLX_WS_OPT_ITEMS, null );
	if ( ! is_array( $items ) || empty( $items ) ) {
		$items = clx_ws_default_items();
		update_option( CLX_WS_OPT_ITEMS, $items );
	}
	return $items;
}

add_action( 'admin_post_clx_ws_save_settings', 'clx_ws_save_settings' );
function clx_ws_save_settings() {
	if ( ! current_user_can( 'manage_options' ) ) { wp_die( 'دسترسی ندارید.' ); }
	check_admin_referer( CLX_WS_NONCE );

	$settings = array(
		'enabled'        => isset( $_POST['enabled'] ) ? 1 : 0,
		'delay'          => max( 0, intval( $_POST['delay'] ?? 7 ) ),
		'frequency'      => sanitize_text_field( $_POST['frequency'] ?? 'once_per_day' ),
		'title'          => sanitize_text_field( $_POST['title'] ?? '' ),
		'subtitle'       => sanitize_text_field( $_POST['subtitle'] ?? '' ),
		'button_text'    => sanitize_text_field( $_POST['button_text'] ?? '' ),
		'closed_text'    => sanitize_text_field( $_POST['closed_text'] ?? '' ),
		'footer_note'    => sanitize_text_field( $_POST['footer_note'] ?? '' ),
		'show_close_btn' => isset( $_POST['show_close_btn'] ) ? 1 : 0,
	);
	update_option( CLX_WS_OPT_SETTINGS, $settings );

	wp_safe_redirect( admin_url( 'admin.php?page=clx-wheel&tab=settings&updated=1' ) );
	exit;
}

add_action( 'admin_post_clx_ws_save_item', 'clx_ws_save_item' );
function clx_ws_save_item() {
	if ( ! current_user_can( 'manage_options' ) ) { wp_die( 'دسترسی ندارید.' ); }
	check_admin_referer( CLX_WS_NONCE );

	$items = clx_ws_get_items();
	$id    = sanitize_text_field( $_POST['item_id'] ?? '' );

	$new_item = array(
		'id'     => $id ? $id : uniqid( 'i', false ),
		'label'  => sanitize_text_field( $_POST['label'] ?? '' ),
		'prize'  => sanitize_text_field( $_POST['prize'] ?? '' ),
		'weight' => max( 1, intval( $_POST['weight'] ?? 10 ) ),
		'color'  => sanitize_hex_color( $_POST['color'] ?? '#0EA5A4' ),
		'active' => isset( $_POST['active'] ) ? 1 : 0,
	);

	$found = false;
	foreach ( $items as $k => $it ) {
		if ( $it['id'] === $new_item['id'] ) {
			$items[ $k ] = $new_item;
			$found       = true;
			break;
		}
	}
	if ( ! $found ) { $items[] = $new_item; }

	update_option( CLX_WS_OPT_ITEMS, $items );
	wp_safe_redirect( admin_url( 'admin.php?page=clx-wheel&tab=items&updated=1' ) );
	exit;
}

add_action( 'admin_post_clx_ws_delete_item', 'clx_ws_delete_item' );
function clx_ws_delete_item() {
	if ( ! current_user_can( 'manage_options' ) ) { wp_die( 'دسترسی ندارید.' ); }
	check_admin_referer( CLX_WS_NONCE );

	$id    = sanitize_text_field( $_GET['id'] ?? '' );
	$items = clx_ws_get_items();
	$items = array_values( array_filter( $items, function( $it ) use ( $id ) {
		return $it['id'] !== $id;
	} ) );
	update_option( CLX_WS_OPT_ITEMS, $items );

	wp_safe_redirect( admin_url( 'admin.php?page=clx-wheel&tab=items&deleted=1' ) );
	exit;
}

	<script>
	(function(){
		var CLX = <?php echo wp_json_encode( $data ); ?>;

		function clxShouldShow(){
			var freq = CLX.settings.frequency;
			var now = Date.now();
			if (freq === 'every_visit') { return true; }
			if (freq === 'once_per_session') {
				return !sessionStorage.getItem('clx_ws_shown');
			}
			if (freq === 'once_per_day') {
				var last = localStorage.getItem('clx_ws_shown_at');
				if (!last) return true;
				return (now - parseInt(last,10)) > 24*60*60*1000;
			}
			if (freq === 'once_forever') {
				return !localStorage.getItem('clx_ws_shown_forever');
			}
			return true;
		}
		function clxMarkShown(){
			sessionStorage.setItem('clx_ws_shown','1');
			localStorage.setItem('clx_ws_shown_at', Date.now().toString());
			localStorage.setItem('clx_ws_shown_forever','1');
		}

		if (!clxShouldShow()) { return; }

		var overlay = document.getElementById('clx-ws-overlay');
		var delay = (CLX.settings.delay || 0) * 1000;

		setTimeout(function(){
			overlay.classList.add('clx-show');
			clxMarkShown();
		}, delay);

		var closeBtn = document.getElementById('clx-ws-close');
		if (closeBtn) {
			closeBtn.addEventListener('click', function(){
				overlay.classList.remove('clx-show');
			});
		}
		overlay.addEventListener('click', function(e){
			if (e.target === overlay) { overlay.classList.remove('clx-show'); }
		});

		/* ---- رسم گردونه روی canvas ---- */
		var items = CLX.items;
		var n = items.length;
		var canvas = document.getElementById('clx-ws-wheel-canvas');
		var ctx = canvas.getContext('2d');
		var cx = canvas.width/2, cy = canvas.height/2, r = canvas.width/2 - 4;
		var sliceAngle = (2*Math.PI) / n;

		function drawWheel(){
			ctx.clearRect(0,0,canvas.width,canvas.height);
			for (var i=0;i<n;i++){
				var start = -Math.PI/2 + i*sliceAngle;
				var end = start + sliceAngle;
				ctx.beginPath();
				ctx.moveTo(cx,cy);
				ctx.arc(cx,cy,r,start,end);
				ctx.closePath();
				ctx.fillStyle = items[i].color || '#0EA5A4';
				ctx.fill();
				ctx.strokeStyle = 'rgba(255,248,236,.35)';
				ctx.lineWidth = 1.5;
				ctx.stroke();

				ctx.save();
				ctx.translate(cx,cy);
				ctx.rotate(start + sliceAngle/2);
				ctx.textAlign = 'right';
				ctx.fillStyle = '#FFF8EC';
				ctx.font = '600 13px Vazirmatn, Tahoma, sans-serif';
				ctx.fillText(items[i].label, r - 12, 4);
				ctx.restore();
			}
		}
		drawWheel();

		/* ---- منطق چرخش و برد وزنی ---- */
		var spinBtn = document.getElementById('clx-ws-spin-btn');
		var spinView = document.getElementById('clx-ws-spin-view');
		var resultView = document.getElementById('clx-ws-result');
		var resultLabel = document.getElementById('clx-ws-result-label');
		var resultCode = document.getElementById('clx-ws-result-code');
		var spun = false;

		function pickWeightedIndex(){
			var total = items.reduce(function(s,it){return s+parseInt(it.weight||1,10);},0);
			var r0 = Math.random()*total;
			var acc = 0;
			for (var i=0;i<n;i++){
				acc += parseInt(items[i].weight||1,10);
				if (r0 <= acc) return i;
			}
			return n-1;
		}

		spinBtn.addEventListener('click', function(){
			if (spun) return;
			spun = true;
			spinBtn.disabled = true;

			var idx = pickWeightedIndex();
			var centerDeg = (idx*sliceAngle + sliceAngle/2) * (180/Math.PI);
			var jitter = (Math.random()-0.5) * (360/n) * 0.5;
			var finalRotation = 6*360 + (360 - centerDeg) + jitter;

			canvas.style.transition = 'transform 4.2s cubic-bezier(.12,.72,.13,1)';
			canvas.style.transform = 'rotate(' + finalRotation + 'deg)';

			setTimeout(function(){
				var won = items[idx];
				resultLabel.textContent = won.label;
				if (won.prize) {
					resultCode.style.display = 'inline-block';
					resultCode.textContent = won.prize;
				} else {
					resultCode.style.display = 'none';
				}
				spinView.style.display = 'none';
				resultView.classList.add('clx-active');
			}, 4300);
		});
	})();
	</script>
