if ( ! function_exists( 'mrd_get_main_category_link' ) ) {
    function mrd_get_main_category_link( $product_id ) {
        $primary_term_id = get_post_meta( $product_id, '_yoast_wpseo_primary_product_cat', true );
        if ( $primary_term_id ) {
            $term = get_term( $primary_term_id, 'product_cat' );
            if ( $term && ! is_wp_error( $term ) ) {
                $link = get_term_link( $term, 'product_cat' );
                if ( ! is_wp_error( $link ) ) return $link;
            }
        }
        $terms = get_the_terms( $product_id, 'product_cat' );
        if ( $terms && ! is_wp_error( $terms ) && ! empty( $terms ) ) {
            $link = get_term_link( $terms[0], 'product_cat' );
            if ( ! is_wp_error( $link ) ) return $link;
        }
        return false;
    }
}

if ( ! function_exists( 'mrd_output_product_data' ) ) {
    function mrd_output_product_data() {
        if ( ! class_exists( 'WooCommerce' ) ) return;


        if ( ! function_exists( 'is_product' ) || ! is_product() ) return;

        global $product;
        if ( ! $product instanceof WC_Product ) {
            $product = wc_get_product( get_the_ID() );
        }
        if ( ! $product instanceof WC_Product || $product->is_in_stock() ) return;

        $link = mrd_get_main_category_link( $product->get_id() );
        if ( ! $link ) return;

        $map = array( $product->get_id() => esc_url_raw( $link ) );

        echo '<script id="mrd-similar-data">window.mrdSimilarMap = ' . wp_json_encode( $map ) . ';</script>';
    }
}

add_action( 'wp_footer', 'mrd_output_product_data', 5 );

if ( ! function_exists( 'mrd_position_script' ) ) {
    function mrd_position_script() {
        if ( ! class_exists( 'WooCommerce' ) ) return;
        ?>
        <script>
        (function () {

            function mrdCreateWrap(link) {
                var wrap = document.createElement('div');
                wrap.className = 'mrd-similar-wrap';
                wrap.setAttribute('data-mrd-btn', '1');

                var a = document.createElement('a');
                a.href = link;
                a.className = 'mrd-similar-products-btn';
                a.innerHTML = '<span class="mrd-similar-icon">⟳</span><span>مشاهده محصولات مشابه</span>';

                wrap.appendChild(a);
                return wrap;
            }

            function mrdFindProductId(el) {
                var node = el;
                var depth = 0;
                while (node && depth < 10) {
                    if (node.className && typeof node.className === 'string') {
                        var m1 = node.className.match(/(?:^|\s)post-(\d+)(?:\s|$)/);
                        if (m1) return m1[1];
                    }
                    if (node.getAttribute) {
                        var dataId = node.getAttribute('data-product_id') || node.getAttribute('data-product-id');
                        if (dataId && /^\d+$/.test(dataId)) return dataId;
                    }
                    node = node.parentElement;
                    depth++;
                }
                return null;
            }

            function mrdFindStockElements() {
                var selectors = [
                    '.stock.out-of-stock',
                    'p.stock.out-of-stock',
                    '.out-of-stock',
                    '.stock',
                    '.wc-block-components-product-badge',
                    '.wc-block-components-product-out-of-stock-badge',
                    '.wc-block-grid__product-onsale + .wc-block-components-product-badge',
                    '.out-of-stock-badge',
                    '.sold-out',
                    '.soldout',
                    '.badge-out-of-stock',
                    '[class*="outofstock"]',
                    '[class*="out-of-stock"]'
                ];

                var found = [];
                var seen = [];

                selectors.forEach(function (sel) {
                    try {
                        document.querySelectorAll(sel).forEach(function (el) {
                            if (seen.indexOf(el) === -1) {
                                seen.push(el);
                                found.push(el);
                            }
                        });
                    } catch (e) {  }
                });

                if (found.length === 0) {
                    var keywords = ['ناموجود', 'نا موجود', 'تمام شد', 'اتمام موجودی', 'موجود نیست', 'out of stock', 'sold out'];
                    var candidates = document.querySelectorAll('p, span, div');
                    candidates.forEach(function (el) {
                        var text = (el.textContent || '').trim();
                        if (text.length > 0 && text.length < 30 && el.children.length <= 2) {
                            var lower = text.toLowerCase();
                            for (var i = 0; i < keywords.length; i++) {
                                if (text.indexOf(keywords[i]) !== -1 || lower.indexOf(keywords[i].toLowerCase()) !== -1) {
                                    found.push(el);
                                    break;
                                }
                            }
                        }
                    });
                }

                return found;
            }

            function mrdFindAddToCartFallback() {
                var selectors = [
                    'a.add_to_cart_button.disabled',
                    'button.disabled[disabled]',
                    '.single_add_to_cart_button[disabled]',
                    'form.cart'
                ];
                var found = [];
                selectors.forEach(function (sel) {
                    try {
                        var els = document.querySelectorAll(sel);
                        if (els.length) found.push(els[els.length - 1]);
                    } catch (e) {}
                });
                return found;
            }

            function mrdAttach(el, link) {
                if (el.dataset.mrdChecked) return;
                el.dataset.mrdChecked = '1';

                if (el.nextElementSibling && el.nextElementSibling.getAttribute && el.nextElementSibling.getAttribute('data-mrd-btn') === '1') {
                    return;
                }

                var wrap = mrdCreateWrap(link);
                el.insertAdjacentElement('afterend', wrap);
            }

            function mrdRun() {
                if (!window.mrdSimilarMap) return;

                var mapKeys = Object.keys(window.mrdSimilarMap);
                if (mapKeys.length === 0) return;

                var stockEls = mrdFindStockElements();
                var singleProductLink = mapKeys.length === 1 ? window.mrdSimilarMap[mapKeys[0]] : null;
                var anyAttached = false;

                stockEls.forEach(function (el) {
                    if (el.dataset.mrdChecked) { anyAttached = true; return; }

                    var link = null;
                    var productId = mrdFindProductId(el);

                    if (productId && window.mrdSimilarMap[productId]) {
                        link = window.mrdSimilarMap[productId];
                    } else if (singleProductLink) {
                        link = singleProductLink;
                    }

                    if (!link) return;
                    mrdAttach(el, link);
                    anyAttached = true;
                });

                if (!anyAttached && singleProductLink) {
                    mrdFindAddToCartFallback().forEach(function (el) {
                        mrdAttach(el, singleProductLink);
                    });
                }
            }

            if (document.readyState === 'loading') {
                document.addEventListener('DOMContentLoaded', mrdRun);
            } else {
                mrdRun();
            }

            setTimeout(mrdRun, 400);
            setTimeout(mrdRun, 1200);
            setTimeout(mrdRun, 2500);
            setTimeout(mrdRun, 5000);

            var mrdThrottle = null;
            var mrdObserver = new MutationObserver(function () {
                if (mrdThrottle) return;
                mrdThrottle = setTimeout(function () {
                    mrdRun();
                    mrdThrottle = null;
                }, 300);
            });
            mrdObserver.observe(document.body, { childList: true, subtree: true });

        })();
        </script>
        <?php
    }
}
