add_action( 'template_redirect', 'wpmc_maybe_show_maintenance_page' );
function wpmc_maybe_show_maintenance_page() {

    $o = wpmc_get_options();

    if ( empty( $o['enabled'] ) ) return;

    // مدیر سایت و درخواست‌های ادمین/اجاکس از حالت تعمیر مستثنی هستند
    if ( current_user_can( 'manage_options' ) ) return;
    if ( defined( 'DOING_AJAX' ) && DOING_AJAX ) return;
    if ( is_admin() ) return;

    // هدر ۵۰۳ برای موتورهای جستجو (توصیه سئو)
    status_header( 503 );
    header( 'Retry-After: 3600' );

    $end_ts = 0;
    if ( ! empty( $o['show_timer'] ) ) {
        $end_ts = wpmc_resolve_end_timestamp( $o['end_datetime'] );
    }

    wpmc_render_maintenance_html( $o, $end_ts );
    exit;
}

/**
 * تبدیل مقدار فیلد «تاریخ و ساعت بازگشایی» به یک timestamp قابل‌اتکا.
 *
 * نکته‌ی مهم: مقدار input[type=datetime-local] هیچ منطقه‌ی زمانی‌ای همراه ندارد
 * (مثلاً "2026-07-20T18:00")، یعنی همان چیزی است که مدیر سایت روی ساعت خودش دیده.
 * قبلاً این رشته مستقیم به strtotime() داده می‌شد که آن را بر اساس تنظیم پیش‌فرض
 * PHP سرور (اغلب UTC) تفسیر می‌کرد، نه منطقه‌ی زمانیِ سایت — همین باعث می‌شد
 * زمان انتخابی «قبول» نشود و شمارش معکوس چند ساعت جابه‌جا نمایش داده شود.
 * اینجا آن را صریحاً با منطقه‌ی زمانیِ ثبت‌شده در پیشخوان ← تنظیمات ← عمومی می‌خوانیم.
 */
function wpmc_resolve_end_timestamp( $raw_datetime ) {
    $fallback = time() + 2 * DAY_IN_SECONDS;

    if ( empty( $raw_datetime ) ) {
        return $fallback;
    }

    try {
        $tz = function_exists( 'wp_timezone' ) ? wp_timezone() : new DateTimeZone( 'UTC' );
        $dt = new DateTime( $raw_datetime, $tz );
        return $dt->getTimestamp();
    } catch ( Exception $e ) {
        return $fallback;
    }
}

/**
 * تولید خطوط مدرج (تیک‌های) لبه‌ی اسطرلاب.
 * این تابع به‌جای نوشتن دستی ۲۴ خط SVG، آن‌ها را حلقه‌ای می‌سازد؛
 * دقیقاً مثل مدرج‌بندی «حلقه‌ی حد» (limb) روی یک اسطرلاب واقعی.
 */
function wpmc_astrolabe_ticks() {
    $out = '';
    for ( $i = 0; $i < 24; $i++ ) {
        $deg      = $i * 15;
        $is_major = ( $i % 6 === 0 );
        $y1       = $is_major ? 4 : 6.5;
        $width    = $is_major ? 1.6 : 0.8;
        $opacity  = $is_major ? 0.95 : 0.4;
        $out .= '<line x1="50" y1="' . $y1 . '" x2="50" y2="11" stroke="var(--wpmc-gold)" stroke-width="' . $width . '" opacity="' . $opacity . '" transform="rotate(' . $deg . ' 50 50)"/>';
    }
    return $out;
}

/**
 * تبدیل اعداد انگلیسی به فارسی برای نمایش در صفحه.
 */
function wpmc_to_fa_digits( $num ) {
    $fa = array( '۰','۱','۲','۳','۴','۵','۶','۷','۸','۹' );
    return preg_replace_callback( '/[0-9]/', function( $m ) use ( $fa ) {
        return $fa[ (int) $m[0] ];
    }, (string) $num );
}
