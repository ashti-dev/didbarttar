	if ( ! function_exists( 'wgp_sanitize_options' ) ) {
		function wgp_sanitize_options( $input ) {
			$defaults = wgp_default_options();
			$out      = array();

			$out['enabled']       = ( isset( $input['enabled'] ) && $input['enabled'] === 'yes' ) ? 'yes' : 'no';
			$out['platform']      = ( isset( $input['platform'] ) && $input['platform'] === 'telegram' ) ? 'telegram' : 'whatsapp';
			$out['contact']       = isset( $input['contact'] ) ? sanitize_text_field( $input['contact'] ) : $defaults['contact'];
			$out['title']         = isset( $input['title'] ) ? sanitize_text_field( $input['title'] ) : $defaults['title'];
			$out['message']       = isset( $input['message'] ) ? sanitize_textarea_field( $input['message'] ) : $defaults['message'];
			$out['button_text']   = isset( $input['button_text'] ) ? sanitize_text_field( $input['button_text'] ) : $defaults['button_text'];
			$out['msg_template']  = isset( $input['msg_template'] ) ? sanitize_text_field( $input['msg_template'] ) : $defaults['msg_template'];
			$out['show_after']    = isset( $input['show_after'] ) ? max( 2, absint( $input['show_after'] ) ) : $defaults['show_after'];
			$out['delay_ms']      = isset( $input['delay_ms'] ) ? max( 0, absint( $input['delay_ms'] ) ) : $defaults['delay_ms'];
			$out['primary_color'] = isset( $input['primary_color'] ) ? sanitize_hex_color( $input['primary_color'] ) : $defaults['primary_color'];
			$out['popup_bg']      = isset( $input['popup_bg'] ) ? sanitize_hex_color( $input['popup_bg'] ) : $defaults['popup_bg'];
			$out['text_color']    = isset( $input['text_color'] ) ? sanitize_hex_color( $input['text_color'] ) : $defaults['text_color'];

			return $out;
		}
	}

	if ( ! function_exists( 'wgp_enqueue_frontend' ) ) {
		add_action( 'wp_enqueue_scripts', 'wgp_enqueue_frontend' );
		function wgp_enqueue_frontend() {

			if ( ! function_exists( 'is_product' ) || ! is_product() ) {
				return; // فقط صفحه محصول
			}

			$o = wgp_get_options();

			if ( $o['enabled'] !== 'yes' || empty( $o['contact'] ) ) {
				return;
			}

			global $post;
			$product_name = $post ? get_the_title( $post ) : '';

			if ( $o['platform'] === 'whatsapp' ) {
				$number = preg_replace( '/[^0-9]/', '', $o['contact'] );
				$text   = str_replace( '{product_name}', $product_name, $o['msg_template'] );
				$link   = 'https://wa.me/' . $number . '?text=' . rawurlencode( $text );
			} else {
				$username = ltrim( $o['contact'], '@' );
				$text     = str_replace( '{product_name}', $product_name, $o['msg_template'] );
				$link     = 'https://t.me/' . rawurlencode( $username ) . '?text=' . rawurlencode( $text );
			}

			// یک هندل خالی برای اتصال CSS/JS inline
			wp_register_style( 'wgp-popup-style', false );
			wp_enqueue_style( 'wgp-popup-style' );
			wp_add_inline_style( 'wgp-popup-style', wgp_build_css( $o ) );

			wp_register_script( 'wgp-popup-script', false, array(), false, true );
			wp_enqueue_script( 'wgp-popup-script' );

			wp_localize_script( 'wgp-popup-script', 'WGP_DATA', array(
				'productId'  => get_the_ID(),
				'showAfter'  => (int) $o['show_after'],
				'delay'      => (int) $o['delay_ms'],
				'link'       => esc_url_raw( $link ),
				'title'      => $o['title'],
				'message'    => $o['message'],
				'buttonText' => $o['button_text'],
				'isRtl'      => is_rtl(),
			) );

			wp_add_inline_script( 'wgp-popup-script', wgp_build_js(), 'after' );
		}
	}

	if ( ! function_exists( 'wgp_build_js' ) ) {
		function wgp_build_js() {
			return "
			(function(){
				try{
					var d = window.WGP_DATA;
					if(!d || !d.productId) return;

					var key = 'wgp_views_' + d.productId;
					var count = parseInt(localStorage.getItem(key) || '0', 10) + 1;
					localStorage.setItem(key, String(count));

					var every = Math.max(2, parseInt(d.showAfter,10) || 3);
					if(count % every !== 0) return; // فقط در بازدید سوم/ششم/نهم و ... نمایش بده

					function buildPopup(){
						var overlay = document.createElement('div');
						overlay.className = 'wgp-overlay';
						overlay.setAttribute('dir', d.isRtl ? 'rtl' : 'ltr');

						var box = document.createElement('div');
						box.className = 'wgp-box';

						var closeBtn = document.createElement('button');
						closeBtn.className = 'wgp-close';
						closeBtn.setAttribute('aria-label','close');
						closeBtn.innerHTML = '&times;';

						var iconWrap = document.createElement('div');
						iconWrap.className = 'wgp-icon-wrap';
						iconWrap.innerHTML = '<svg viewBox=\"0 0 24 24\"><path d=\"M12 2C6.48 2 2 6.02 2 11c0 2.25 1.05 4.3 2.78 5.88L4 22l5.3-1.63C10.16 20.78 11.06 21 12 21c5.52 0 10-4.02 10-9s-4.48-10-10-10z\"/></svg>';

						var title = document.createElement('p');
						title.className = 'wgp-title';
						title.textContent = d.title;

						var msg = document.createElement('p');
						msg.className = 'wgp-message';
						msg.textContent = d.message;

						var btn = document.createElement('a');
						btn.className = 'wgp-btn';
						btn.href = d.link;
						btn.target = '_blank';
						btn.rel = 'noopener noreferrer nofollow';
						btn.textContent = d.buttonText;

						box.appendChild(closeBtn);
						box.appendChild(iconWrap);
						box.appendChild(title);
						box.appendChild(msg);
						box.appendChild(btn);
						overlay.appendChild(box);
						document.body.appendChild(overlay);

						function closePopup(){
							overlay.classList.remove('wgp-show');
							setTimeout(function(){ overlay.remove(); }, 350);
						}
						closeBtn.addEventListener('click', closePopup);
						overlay.addEventListener('click', function(e){
							if(e.target === overlay) closePopup();
						});
						document.addEventListener('keydown', function esc(e){
							if(e.key === 'Escape'){ closePopup(); document.removeEventListener('keydown', esc); }
						});

						requestAnimationFrame(function(){
							requestAnimationFrame(function(){ overlay.classList.add('wgp-show'); });
						});
					}

					var delay = parseInt(d.delay,10) || 0;
					if(document.readyState === 'complete' || document.readyState === 'interactive'){
						setTimeout(buildPopup, delay);
					} else {
						document.addEventListener('DOMContentLoaded', function(){
							setTimeout(buildPopup, delay);
						});
					}
				}catch(e){ /* fail silently, never break the shop */ }
			})();
			";
		}
	}
